# P4.1 — Framework-Free REST API

**Phase:** 4 — Networking & HTTP
**Topic:** Router, middleware, validation, CORS, pagination, JSON file persistence — all on `node:http`
**Type:** Project

---

## Goal

Build a complete **Bookmarks REST API** without any framework. You'll reuse your router and middleware ideas from A4.2 and add input validation, CORS, pagination, filtering, conditional requests, and a persistent JSON file store with atomic writes and a write queue.

## What to Build

- `GET /bookmarks?tag=&q=&limit=&offset=` — list with filtering, search, and pagination (`X-Total-Count` header)
- `GET /bookmarks/:id` — one bookmark, with an `ETag`; `If-None-Match` returns `304`
- `POST /bookmarks` — create (`url`, `title`, optional `tags[]`)
- `PATCH /bookmarks/:id` — partial update
- `DELETE /bookmarks/:id`
- `GET /health` — status and uptime
- Validation with field-level error messages (`422`)
- CORS for a configurable origin, including `OPTIONS` preflight
- Consistent JSON error format: `{ error: { status, message, details?, requestId } }`
- Data persisted to `data/bookmarks.json`; concurrent writes are serialised
- Request logging and request IDs
- Graceful shutdown on `SIGTERM` / `SIGINT`

### Validation

```js
// src/lib/validate.js
import { HttpError } from './errors.js';

const rules = {
  string: (v, o) => typeof v === 'string' && v.trim().length >= (o.min ?? 1) && v.length <= (o.max ?? Infinity),
  url: (v) => { try { return ['http:', 'https:'].includes(new URL(v).protocol); } catch { return false; } },
  tags: (v) => Array.isArray(v) && v.length <= 10 && v.every(t => typeof t === 'string' && /^[a-z0-9-]{1,30}$/.test(t)),
};

export function validate(body, schema, { partial = false } = {}) {
  const details = {};
  const out = {};
  for (const [field, spec] of Object.entries(schema)) {
    const value = body[field];
    if (value === undefined) {
      if (spec.required && !partial) details[field] = 'is required';
      continue;
    }
    if (!rules[spec.type](value, spec)) details[field] = spec.message;
    else out[field] = typeof value === 'string' ? value.trim() : value;
  }
  for (const key of Object.keys(body)) if (!(key in schema)) details[key] = 'is not allowed';
  if (Object.keys(details).length) throw new HttpError(422, 'validation failed', {}, details);
  return out;
}
```

Extend `HttpError` to accept `details` as a fourth argument and include it in error responses.

### CORS

```js
// src/middleware/cors.js
export function cors({ origin = '*', methods = 'GET,POST,PATCH,DELETE,OPTIONS', maxAge = 600 } = {}) {
  return async (ctx, next) => {
    const reqOrigin = ctx.req.headers.origin;
    if (reqOrigin && (origin === '*' || origin === reqOrigin)) {
      ctx.res.setHeader('access-control-allow-origin', origin === '*' ? '*' : reqOrigin);
      ctx.res.setHeader('vary', 'Origin');
      ctx.res.setHeader('access-control-expose-headers', 'x-total-count, x-request-id, etag');
    }
    if (ctx.req.method === 'OPTIONS' && ctx.req.headers['access-control-request-method']) {
      ctx.res.writeHead(204, {
        'access-control-allow-methods': methods,
        'access-control-allow-headers': 'content-type, authorization',
        'access-control-max-age': String(maxAge),
      });
      return ctx.res.end();                 // preflight answered — skip the router
    }
    await next();
  };
}
```

### JSON store with serialised writes

```js
// src/store/jsonStore.js
import { readFile, writeFile, rename, mkdir } from 'node:fs/promises';
import { dirname } from 'node:path';

export class JsonStore {
  #path;
  #data = { nextId: 1, items: [] };
  #queue = Promise.resolve();        // writes run one after another

  constructor(path) { this.#path = path; }

  async load() {
    await mkdir(dirname(this.#path), { recursive: true });
    try {
      this.#data = JSON.parse(await readFile(this.#path, 'utf8'));
    } catch (err) {
      if (err.code !== 'ENOENT') throw err;
    }
    return this;
  }

  all() { return this.#data.items; }
  get(id) { return this.#data.items.find(i => i.id === id); }

  // Mutate in memory synchronously, then queue a save. Returns when the change is on disk.
  mutate(fn) {
    const result = fn(this.#data);
    const snapshot = JSON.stringify(this.#data, null, 2);
    const write = this.#queue
      .catch(() => {})                 // a failed earlier write must not block later ones
      .then(async () => {
        const tmp = `${this.#path}.tmp`;
        await writeFile(tmp, snapshot);
        await rename(tmp, this.#path);
      });
    this.#queue = write;
    return write.then(() => result);   // the caller sees its own write's success or failure
  }

  flush() { return this.#queue.catch(() => {}); }
}
```

Two simultaneous `mutate()` calls can never interleave their file writes, each write uses the snapshot taken right after its own change, and one failed write doesn't poison the queue.

### Routes

```js
// src/routes/bookmarks.js
import { createHash } from 'node:crypto';
import { Router } from '../lib/router.js';
import { HttpError } from '../lib/errors.js';
import { validate } from '../lib/validate.js';

const SCHEMA = {
  url: { type: 'url', required: true, message: 'must be an http(s) URL' },
  title: { type: 'string', required: true, max: 200, message: 'must be 1–200 characters' },
  tags: { type: 'tags', message: 'must be up to 10 lowercase tags (a-z, 0-9, -)' },
};

const etag = (obj) => `"${createHash('sha1').update(JSON.stringify(obj)).digest('base64url')}"`;

function intParam(raw, fallback, { min, max }) {
  if (raw === null) return fallback;
  const n = Number(raw);
  if (!Number.isInteger(n) || n < min || n > max) throw new HttpError(400, `expected integer ${min}–${max}, got "${raw}"`);
  return n;
}

export function bookmarkRoutes(store) {
  const find = (id) => {
    const b = store.get(Number(id));
    if (!b) throw new HttpError(404, `bookmark ${id} not found`);
    return b;
  };

  return new Router()
    .get('/bookmarks', (ctx) => {
      const q = ctx.url.searchParams;
      const limit = intParam(q.get('limit'), 20, { min: 1, max: 100 });
      const offset = intParam(q.get('offset'), 0, { min: 0, max: Number.MAX_SAFE_INTEGER });
      const tag = q.get('tag');
      const search = q.get('q')?.toLowerCase();

      let items = store.all();
      if (tag) items = items.filter(b => b.tags.includes(tag));
      if (search) items = items.filter(b => b.title.toLowerCase().includes(search) || b.url.includes(search));

      ctx.res.setHeader('x-total-count', String(items.length));
      ctx.json(items.slice(offset, offset + limit));
    })
    .get('/bookmarks/:id', (ctx) => {
      const b = find(ctx.params.id);
      const tag = etag(b);
      if (ctx.req.headers['if-none-match'] === tag) return ctx.status(304);
      ctx.res.setHeader('etag', tag);
      ctx.json(b);
    })
    .post('/bookmarks', async (ctx) => {
      const input = validate(ctx.body, SCHEMA);
      const bookmark = await store.mutate((data) => {
        const now = new Date().toISOString();
        const b = { id: data.nextId++, tags: [], ...input, createdAt: now, updatedAt: now };
        data.items.push(b);
        return b;
      });
      ctx.res.setHeader('location', `/bookmarks/${bookmark.id}`);
      ctx.json(bookmark, 201);
    })
    .patch('/bookmarks/:id', async (ctx) => {
      const existing = find(ctx.params.id);
      const changes = validate(ctx.body, SCHEMA, { partial: true });
      const updated = await store.mutate(() => Object.assign(existing, changes, { updatedAt: new Date().toISOString() }));
      ctx.json(updated);
    })
    .delete('/bookmarks/:id', async (ctx) => {
      const existing = find(ctx.params.id);
      await store.mutate((data) => { data.items = data.items.filter(b => b !== existing); });
      ctx.status(204);
    });
}
```

### App and server

```js
// src/app.js
import { compose } from './lib/compose.js';
import { requestId } from './middleware/requestId.js';
import { logger } from './middleware/logger.js';
import { cors } from './middleware/cors.js';
import { jsonBody } from './middleware/jsonBody.js';
import { bookmarkRoutes } from './routes/bookmarks.js';
import { HttpError } from './lib/errors.js';

export function createApp(store, { corsOrigin }) {
  const router = bookmarkRoutes(store);
  router.get('/health', (ctx) => ctx.json({ status: 'ok', uptime: process.uptime() }));

  const pipeline = compose([
    requestId, logger, cors({ origin: corsOrigin }), jsonBody({ limit: 64_000 }),
    async (ctx) => {
      const { handler, params } = router.resolve(ctx.req.method, ctx.url.pathname);
      ctx.params = params;
      await handler(ctx);
    },
  ]);

  return async function handle(req, res) {
    const ctx = {
      req, res, url: new URL(req.url, 'http://localhost'),
      json(data, status = 200) { res.writeHead(status, { 'content-type': 'application/json' }); res.end(JSON.stringify(data)); },
      status(code) { res.writeHead(code); res.end(); },
    };
    try {
      await pipeline(ctx);
    } catch (err) {
      const known = err instanceof HttpError;
      if (!known) console.error(err);
      if (res.headersSent) return res.destroy();
      const status = known ? err.status : 500;
      res.writeHead(status, { 'content-type': 'application/json', ...(err.headers ?? {}) });
      res.end(JSON.stringify({ error: { status, message: known ? err.message : 'internal server error', details: err.details, requestId: ctx.id } }));
    }
  };
}
```

```js
// src/server.js
import { createServer } from 'node:http';
import { JsonStore } from './store/jsonStore.js';
import { createApp } from './app.js';

const port = Number(process.env.PORT ?? 3000);
const store = await new JsonStore(process.env.DATA_FILE ?? 'data/bookmarks.json').load();
const server = createServer(createApp(store, { corsOrigin: process.env.CORS_ORIGIN ?? '*' }));

server.listen(port, () => console.log(`bookmarks api on :${port}`));

async function shutdown(signal) {
  console.log(`${signal} received, shutting down`);
  server.close();                     // stop accepting new connections
  server.closeIdleConnections();
  const force = setTimeout(() => server.closeAllConnections(), 10_000).unref();
  await store.flush();                // make sure pending writes reach disk
  clearTimeout(force);
}
process.once('SIGTERM', shutdown);
process.once('SIGINT', shutdown);
```
