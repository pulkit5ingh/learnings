# P2.2 — Event-Driven Task Runner

**Phase:** 2 — Asynchronous Programming
**Topic:** EventEmitter subclasses, dependency graphs, parallel execution, cycle detection, lifecycle events
**Type:** Project

---

## Goal

Build a tiny task runner (think a minimal `make` or `turbo`) as an `EventEmitter`. Tasks declare dependencies; the runner executes them in dependency order, runs independent tasks in parallel up to a limit, and emits lifecycle events that a separate reporter turns into console output.

## What to Build

- A `TaskRunner` class extending `EventEmitter`
- `runner.task(name, deps, fn)` registers an async task
- `runner.run(target)` runs `target` and all its transitive dependencies
- Each task runs **once**, only after all its dependencies succeed
- Independent tasks run in parallel, limited by `concurrency`
- Cycle detection before anything runs (`a → b → a` is rejected with a clear message)
- If a task fails, dependents are skipped; unrelated tasks still finish
- Events: `run:start`, `task:start`, `task:done`, `task:fail`, `task:skip`, `run:end`
- A reporter that subscribes to events and prints timing — the runner itself never logs
- A `tasks.js` file defining a demo build pipeline, run via `node runner-cli.js build`

## Architecture

```
task-runner/
├── package.json
├── src/
│   ├── TaskRunner.js     ← graph, scheduling, events
│   └── reporter.js       ← console output from events
├── tasks.js              ← demo task definitions
└── runner-cli.js         ← loads tasks.js and runs a target
```

## Implementation

### TaskRunner

```js
// src/TaskRunner.js
import { EventEmitter } from 'node:events';

export class TaskRunner extends EventEmitter {
  #tasks = new Map();
  #concurrency;

  constructor({ concurrency = 2 } = {}) {
    super();
    this.#concurrency = concurrency;
  }

  task(name, deps, fn) {
    if (this.#tasks.has(name)) throw new Error(`task "${name}" already defined`);
    this.#tasks.set(name, { name, deps, fn });
    return this;
  }

  // Returns tasks needed for `target` in a valid order, or throws on unknown deps / cycles
  #plan(target) {
    const order = [];
    const state = new Map();            // name → 'visiting' | 'done'

    const visit = (name, path) => {
      const task = this.#tasks.get(name);
      if (!task) throw new Error(`unknown task "${name}"${path.length ? ` (needed by ${path.at(-1)})` : ''}`);
      if (state.get(name) === 'done') return;
      if (state.get(name) === 'visiting') {
        throw new Error(`dependency cycle: ${[...path.slice(path.indexOf(name)), name].join(' → ')}`);
      }
      state.set(name, 'visiting');
      for (const dep of task.deps) visit(dep, [...path, name]);
      state.set(name, 'done');
      order.push(task);
    };

    visit(target, []);
    return order;
  }

  async run(target) {
    const plan = this.#plan(target);
    const status = new Map(plan.map(t => [t.name, 'pending']));   // pending|running|done|failed|skipped
    const started = performance.now();
    this.emit('run:start', { target, tasks: plan.map(t => t.name) });

    await new Promise((resolve) => {
      let running = 0;

      const schedule = () => {
        for (const task of plan) {
          if (status.get(task.name) !== 'pending') continue;

          const depStates = task.deps.map(d => status.get(d));
          if (depStates.some(s => s === 'failed' || s === 'skipped')) {
            status.set(task.name, 'skipped');
            this.emit('task:skip', { name: task.name, reason: 'a dependency failed' });
            continue;
          }
          if (!depStates.every(s => s === 'done') || running >= this.#concurrency) continue;

          running++;
          status.set(task.name, 'running');
          this.#execute(task).then((ok) => {
            status.set(task.name, ok ? 'done' : 'failed');
            running--;
            schedule();
          });
        }
        if (running === 0 && ![...status.values()].includes('pending')) resolve();
      };

      schedule();
    });

    const summary = Object.fromEntries(status);
    const ok = Object.values(summary).every(s => s === 'done');
    this.emit('run:end', { target, ok, summary, ms: Math.round(performance.now() - started) });
    return ok;
  }

  async #execute(task) {
    const start = performance.now();
    this.emit('task:start', { name: task.name });
    try {
      await task.fn();
      this.emit('task:done', { name: task.name, ms: Math.round(performance.now() - start) });
      return true;
    } catch (error) {
      this.emit('task:fail', { name: task.name, error, ms: Math.round(performance.now() - start) });
      return false;
    }
  }
}
```

Note how scheduling re-runs every time a task settles: that is the event-driven core. `schedule()` runs synchronously between awaits, so there are no race conditions on `running` or `status`.

### Reporter

```js
// src/reporter.js
export function attachReporter(runner, out = process.stderr) {
  const log = (line) => out.write(`${line}\n`);

  runner.on('run:start', ({ target, tasks }) => log(`▶ ${target}: ${tasks.join(', ')}`));
  runner.on('task:start', ({ name }) => log(`  … ${name}`));
  runner.on('task:done', ({ name, ms }) => log(`  ✔ ${name} (${ms}ms)`));
  runner.on('task:fail', ({ name, error, ms }) => log(`  ✖ ${name} (${ms}ms): ${error.message}`));
  runner.on('task:skip', ({ name, reason }) => log(`  ⊘ ${name} skipped — ${reason}`));
  runner.on('run:end', ({ ok, ms }) => log(`${ok ? '✔ success' : '✖ failed'} in ${ms}ms`));
}
```

### Demo tasks

```js
// tasks.js
import { setTimeout as sleep } from 'node:timers/promises';

export default function define(runner) {
  runner
    .task('clean', [], () => sleep(100))
    .task('lint', ['clean'], () => sleep(300))
    .task('compile', ['clean'], () => sleep(400))
    .task('test', ['compile'], async () => {
      await sleep(300);
      if (process.env.FAIL_TESTS) throw new Error('3 tests failed');
    })
    .task('docs', ['compile'], () => sleep(200))
    .task('bundle', ['compile'], () => sleep(250))
    .task('build', ['lint', 'test', 'bundle', 'docs'], () => sleep(50));
}
```

### CLI

```js
// runner-cli.js
import { parseArgs } from 'node:util';
import { TaskRunner } from './src/TaskRunner.js';
import { attachReporter } from './src/reporter.js';
import define from './tasks.js';

const { values, positionals } = parseArgs({
  options: { concurrency: { type: 'string', short: 'c', default: '2' } },
  allowPositionals: true,
});

const runner = new TaskRunner({ concurrency: Number(values.concurrency) });
attachReporter(runner);
define(runner);

try {
  const ok = await runner.run(positionals[0] ?? 'build');
  process.exitCode = ok ? 0 : 1;
} catch (err) {
  console.error(err.message);       // unknown task or cycle
  process.exitCode = 2;
}
```

### Try it

```bash
node runner-cli.js build
node runner-cli.js build -c 4          # faster: more parallelism
FAIL_TESTS=1 node runner-cli.js build  # test fails → build skipped, docs/bundle still finish
node runner-cli.js deploy              # unknown task → exit 2
```

Add `.task('clean', ['build'], …)` temporarily (rename the original) to see cycle detection.

## Deliverables

- [ ] `TaskRunner` extends `EventEmitter` and never writes to the console itself
- [ ] Tasks run only after all dependencies succeed, and each runs exactly once
- [ ] Independent tasks run in parallel, never exceeding `concurrency`
- [ ] Cycles and unknown dependencies are reported before any task runs
- [ ] A failing task causes its dependents to be skipped while unrelated tasks still complete
- [ ] All six lifecycle events are emitted with useful payloads
- [ ] Raising concurrency measurably reduces total run time
- [ ] Exit code reflects success (0), task failure (1), or a configuration error (2)
