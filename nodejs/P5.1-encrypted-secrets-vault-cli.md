# P5.1 — Encrypted Secrets Vault CLI

**Phase:** 5 — Security & Cryptography
**Topic:** scrypt key derivation, AES-256-GCM, versioned file format, hidden password input, atomic writes, export to env
**Type:** Project

---

## Goal

Build `vault` — a small password-protected secrets manager (like a minimal `pass` or 1Password CLI). Secrets are stored in a single encrypted file. Every operation asks for the master password, derives a key with scrypt, and decrypts/encrypts the whole vault with AES-256-GCM.

## What to Build

- `vault init` — create a new vault, asking for the master password twice
- `vault set <name> [--value V]` — add or update a secret (prompts with hidden input if `--value` is omitted)
- `vault get <name>` — print a secret to stdout (nothing else on stdout, so it can be piped)
- `vault list` — list names and last-updated timestamps, never values
- `vault rm <name>`
- `vault passwd` — change the master password (re-encrypts with a new salt and key)
- `vault export --format env` — print `NAME=value` lines for `eval` or `.env` generation
- `vault exec -- <cmd> [args…]` — run a command with secrets injected as environment variables
- Master password from a hidden prompt, or `VAULT_PASSWORD` env var for automation
- File location `~/.vault.enc` or `VAULT_FILE`; file mode `0600`
- Clear errors: wrong password vs corrupted file vs missing vault

## Architecture

```
vault-cli/
├── package.json
├── bin/
│   └── vault.js
└── src/
    ├── crypto.js       ← KDF + seal/open with a versioned binary header
    ├── file.js         ← read/write the vault file atomically with 0600
    ├── prompt.js       ← hidden password input
    └── commands.js
```

File format (binary):

```
magic "VLT1"(4) | N exponent(1) | salt(16) | iv(12) | tag(16) | ciphertext(…)
```

The plaintext is JSON: `{ "secrets": { "NAME": { "value": "...", "updatedAt": "..." } } }`.

## Implementation

### Crypto

```js
// src/crypto.js
import { scrypt, randomBytes, createCipheriv, createDecipheriv } from 'node:crypto';
import { promisify } from 'node:util';

const scryptAsync = promisify(scrypt);
const MAGIC = Buffer.from('VLT1');
const HEADER = 4 + 1 + 16 + 12 + 16;
const DEFAULT_LOG_N = 17;   // N = 2^17 ≈ 128 MB, ~0.3–0.5 s

export class WrongPasswordError extends Error { name = 'WrongPasswordError'; }

async function deriveKey(password, salt, logN) {
  const N = 2 ** logN;
  return scryptAsync(password.normalize('NFKC'), salt, 32, { N, r: 8, p: 1, maxmem: 256 * N * 8 });
}

export async function seal(obj, password, logN = DEFAULT_LOG_N) {
  const salt = randomBytes(16);
  const iv = randomBytes(12);
  const key = await deriveKey(password, salt, logN);
  const cipher = createCipheriv('aes-256-gcm', key, iv);
  cipher.setAAD(MAGIC);
  const ct = Buffer.concat([cipher.update(JSON.stringify(obj), 'utf8'), cipher.final()]);
  return Buffer.concat([MAGIC, Buffer.from([logN]), salt, iv, cipher.getAuthTag(), ct]);
}

export async function open(buf, password) {
  if (buf.length < HEADER || !buf.subarray(0, 4).equals(MAGIC)) throw new Error('not a vault file (bad header)');
  const logN = buf[4];
  if (logN < 14 || logN > 20) throw new Error('vault file has invalid KDF parameters');
  const salt = buf.subarray(5, 21);
  const iv = buf.subarray(21, 33);
  const tag = buf.subarray(33, 49);
  const ct = buf.subarray(49);

  const key = await deriveKey(password, salt, logN);
  const decipher = createDecipheriv('aes-256-gcm', key, iv);
  decipher.setAAD(MAGIC);
  decipher.setAuthTag(tag);
  let plaintext;
  try {
    plaintext = Buffer.concat([decipher.update(ct), decipher.final()]);
  } catch {
    // GCM can't distinguish a wrong key from tampering — both fail authentication
    throw new WrongPasswordError('wrong master password (or the vault file was modified)');
  }
  return JSON.parse(plaintext.toString('utf8'));
}
```

### File handling

```js
// src/file.js
import { readFile, writeFile, rename, chmod } from 'node:fs/promises';
import { homedir } from 'node:os';
import { join, dirname, basename } from 'node:path';

export const VAULT_FILE = process.env.VAULT_FILE ?? join(homedir(), '.vault.enc');

export async function readVault() {
  try {
    return await readFile(VAULT_FILE);
  } catch (err) {
    if (err.code === 'ENOENT') throw new Error(`no vault at ${VAULT_FILE} — run "vault init"`);
    throw err;
  }
}

export async function writeVault(buf) {
  const tmp = join(dirname(VAULT_FILE), `.${basename(VAULT_FILE)}.${process.pid}.tmp`);
  await writeFile(tmp, buf, { mode: 0o600 });
  await chmod(tmp, 0o600);                     // in case umask changed the mode
  await rename(tmp, VAULT_FILE);
}
```

### Hidden password prompt

```js
// src/prompt.js
import { stdin, stderr } from 'node:process';

export function promptHidden(question) {
  if (!stdin.isTTY) throw new Error('no TTY for password prompt — set VAULT_PASSWORD');
  return new Promise((resolve, reject) => {
    stderr.write(question);                  // prompt on stderr keeps stdout clean
    stdin.setRawMode(true);
    stdin.resume();
    stdin.setEncoding('utf8');
    let input = '';
    const onData = (ch) => {
      for (const c of ch) {
        if (c === '\r' || c === '\n') return finish();
        if (c === '\u0003') { cleanup(); stderr.write('\n'); return reject(new Error('cancelled')); }  // Ctrl+C
        if (c === '\u007f' || c === '\b') input = input.slice(0, -1);                              // backspace
        else input += c;
      }
    };
    const cleanup = () => { stdin.off('data', onData); stdin.setRawMode(false); stdin.pause(); };
    const finish = () => { cleanup(); stderr.write('\n'); resolve(input); };
    stdin.on('data', onData);
  });
}

export async function getPassword({ confirm = false, label = 'Master password: ' } = {}) {
  if (process.env.VAULT_PASSWORD) return process.env.VAULT_PASSWORD;
  const pw = await promptHidden(label);
  if (confirm) {
    if (pw.length < 12) throw new Error('master password must be at least 12 characters');
    if ((await promptHidden('Confirm password: ')) !== pw) throw new Error('passwords do not match');
  }
  return pw;
}
```

### Commands

```js
// src/commands.js
import { spawn } from 'node:child_process';
import { access } from 'node:fs/promises';
import { seal, open } from './crypto.js';
import { readVault, writeVault, VAULT_FILE } from './file.js';
import { getPassword, promptHidden } from './prompt.js';

const NAME = /^[A-Z_][A-Z0-9_]*$/;

async function unlock() {
  const password = await getPassword();
  return { password, data: await open(await readVault(), password) };
}

export async function init() {
  try { await access(VAULT_FILE); throw new Error(`vault already exists at ${VAULT_FILE}`); }
  catch (err) { if (err.code !== 'ENOENT') throw err; }
  const password = await getPassword({ confirm: true });
  await writeVault(await seal({ secrets: {} }, password));
  console.error(`created ${VAULT_FILE}`);
}

export async function set([name], { value }) {
  if (!NAME.test(name ?? '')) throw new Error('secret names must be UPPER_SNAKE_CASE');
  const { password, data } = await unlock();
  const secret = value ?? (await promptHidden(`Value for ${name}: `));
  data.secrets[name] = { value: secret, updatedAt: new Date().toISOString() };
  await writeVault(await seal(data, password));
  console.error(`saved ${name}`);
}

export async function get([name]) {
  const { data } = await unlock();
  const s = data.secrets[name];
  if (!s) throw new Error(`no secret named ${name}`);
  process.stdout.write(s.value + (process.stdout.isTTY ? '\n' : ''));
}

export async function list() {
  const { data } = await unlock();
  const rows = Object.entries(data.secrets).map(([name, s]) => ({ name, updatedAt: s.updatedAt }));
  rows.length ? console.table(rows) : console.error('vault is empty');
}

export async function rm([name]) {
  const { password, data } = await unlock();
  if (!data.secrets[name]) throw new Error(`no secret named ${name}`);
  delete data.secrets[name];
  await writeVault(await seal(data, password));
  console.error(`removed ${name}`);
}

export async function passwd() {
  const { data } = await unlock();
  const next = await getPassword({ confirm: true, label: 'New master password: ' });
  await writeVault(await seal(data, next));          // new salt + IV + key
  console.error('master password changed');
}

const shellQuote = (v) => `'${v.replaceAll("'", `'\\''`)}'`;

export async function exportCmd(_args, { format = 'env' }) {
  if (format !== 'env') throw new Error('only --format env is supported');
  const { data } = await unlock();
  for (const [name, s] of Object.entries(data.secrets)) console.log(`${name}=${shellQuote(s.value)}`);
}

export async function exec(args) {
  if (!args.length) throw new Error('usage: vault exec -- <command> [args...]');
  const { data } = await unlock();
  const env = { ...process.env, ...Object.fromEntries(Object.entries(data.secrets).map(([k, s]) => [k, s.value])) };
  delete env.VAULT_PASSWORD;                          // don't leak the master password to children
  const child = spawn(args[0], args.slice(1), { stdio: 'inherit', env });
  await new Promise((resolve, reject) => {
    child.on('error', reject);
    child.on('exit', (code, signal) => { process.exitCode = code ?? (signal ? 1 : 0); resolve(); });
  });
}
```

### CLI

```js
#!/usr/bin/env node
// bin/vault.js
import { parseArgs } from 'node:util';
import * as cmd from '../src/commands.js';
import { WrongPasswordError } from '../src/crypto.js';

const COMMANDS = { init: cmd.init, set: cmd.set, get: cmd.get, list: cmd.list, rm: cmd.rm, passwd: cmd.passwd, export: cmd.exportCmd, exec: cmd.exec };

const { values, positionals } = parseArgs({
  options: { value: { type: 'string' }, format: { type: 'string' } },
  allowPositionals: true,
});
const [name, ...rest] = positionals;

try {
  const handler = COMMANDS[name];
  if (!handler) throw new Error(`usage: vault <${Object.keys(COMMANDS).join('|')}> …`);
  await handler(rest, values);
} catch (err) {
  console.error(`vault: ${err.message}`);
  process.exitCode = err instanceof WrongPasswordError ? 3 : 1;
}
```

`parseArgs` stops treating arguments as options after `--`, so `vault exec -- node -e "…"` passes everything through as positionals.

### Try it

```bash
export VAULT_FILE=/tmp/test.vault
node bin/vault.js init
node bin/vault.js set DATABASE_URL
node bin/vault.js set API_KEY --value sk_test_123
node bin/vault.js list
node bin/vault.js get API_KEY | pbcopy                  # nothing but the value on stdout
node bin/vault.js exec -- node -e "console.log(process.env.API_KEY)"
VAULT_PASSWORD=wrong node bin/vault.js list; echo "exit $?"   # 3
ls -l /tmp/test.vault                                   # -rw-------
xxd /tmp/test.vault | head -3                           # magic, no readable secrets
printf 'X' | dd of=/tmp/test.vault bs=1 seek=60 conv=notrunc; node bin/vault.js list   # tampering detected
```

## Deliverables

- [ ] The vault file contains no readable secrets and starts with the `VLT1` magic and KDF parameter byte
- [ ] Each save uses a fresh salt and IV
- [ ] Wrong passwords and tampered files both fail authentication (exit code 3)
- [ ] The master password is read without echoing, or from `VAULT_PASSWORD`
- [ ] Prompts and status messages go to stderr; `get` writes only the value to stdout
- [ ] Writes are atomic and the file mode is `0600`
- [ ] `passwd` re-encrypts under a new key; the old password no longer works
- [ ] `exec` injects secrets into the child's environment without passing `VAULT_PASSWORD`
- [ ] `export` output is safely shell-quoted
