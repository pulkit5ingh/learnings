# P3.1 — Log File Analyzer

**Phase:** 3 — File System & Streams
**Topic:** Line-by-line streaming, gzip decompression, aggregation, top-N, constant-memory processing
**Type:** Project

---

## Goal

Build `loganalyze` — a CLI that streams web server access logs (plain or `.gz`, any size) and produces a report: request counts by status, top paths, top IPs, error rate, latency percentiles, and requests per minute. Memory must stay flat no matter how big the input is.

## What to Build

- `loganalyze access.log [more.log.gz …]` — multiple files, gzip detected by extension, or `-` for stdin
- Parses lines in this format (one JSON object per line):
  `{"ts":"2026-10-30T12:00:01.123Z","ip":"10.0.0.5","method":"GET","path":"/api/users/42","status":200,"ms":37}`
- Malformed lines are counted and skipped, never crash the tool
- Paths are normalised (`/api/users/42` → `/api/users/:id`) so top-N is meaningful
- Report: total, status class counts (2xx/3xx/4xx/5xx), error rate, top 10 paths, top 10 IPs, p50/p95/p99 latency, busiest minute
- `--since` / `--until` ISO timestamps to filter
- `--json` for machine output
- A generator script that creates a multi-million-line test log
- Progress (lines/sec) on stderr

### Test data generator

```js
// scripts/generate.js
import { createWriteStream } from 'node:fs';
import { once } from 'node:events';
import { createGzip } from 'node:zlib';
import { pipeline } from 'node:stream/promises';
import { Readable } from 'node:stream';

const lines = Number(process.argv[2] ?? 2_000_000);
const out = process.argv[3] ?? 'access.log';
const paths = ['/', '/login', '/api/users/:n', '/api/orders/:n', '/static/app.js', '/api/search'];
const statuses = [200, 200, 200, 200, 201, 304, 400, 404, 500, 503];
const pick = (arr) => arr[Math.floor(Math.random() * arr.length)];
const start = Date.parse('2026-10-30T00:00:00Z');

async function* generate() {
  for (let i = 0; i < lines; i++) {
    yield JSON.stringify({
      ts: new Date(start + i * 50).toISOString(),
      ip: `10.0.${Math.floor(Math.random() * 4)}.${Math.floor(Math.random() * 50)}`,
      method: Math.random() < 0.8 ? 'GET' : 'POST',
      path: pick(paths).replace(':n', String(Math.floor(Math.random() * 1000))),
      status: pick(statuses),
      ms: Math.round(Math.random() ** 3 * 2000),
    }) + '\n';
    if (i % 1000 === 0 && Math.random() < 0.01) yield 'garbage line\n';
  }
}

const stages = [Readable.from(generate())];
if (out.endsWith('.gz')) stages.push(createGzip());
stages.push(createWriteStream(out));
await pipeline(...stages);
console.log(`wrote ${lines} lines to ${out}`);
```

### Line source

```js
// src/source.js
import { createReadStream } from 'node:fs';
import { createGunzip } from 'node:zlib';
import { createInterface } from 'node:readline';

export function lines(file) {
  const raw = file === '-' ? process.stdin : createReadStream(file);
  const input = file.endsWith('.gz') ? raw.pipe(createGunzip()) : raw;
  // .pipe() doesn't forward errors: push file errors (e.g. ENOENT) into the gunzip stream
  if (input !== raw) raw.on('error', (err) => input.destroy(err));
  return createInterface({ input, crlfDelay: Infinity });   // async iterable of lines
}
```

`readline` handles lines split across chunks and multi-byte characters for you, and forwards input stream errors so the `for await` loop throws.

### Parsing

```js
// src/parse.js
const NUMERIC = /^\d+$/;
const UUID = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;

export function normalisePath(path) {
  const clean = path.split('?')[0];
  return clean
    .split('/')
    .map(seg => (NUMERIC.test(seg) || UUID.test(seg) ? ':id' : seg))
    .join('/');
}

export function parseLine(line) {
  if (!line || line[0] !== '{') return null;
  try {
    const r = JSON.parse(line);
    if (typeof r.status !== 'number' || typeof r.path !== 'string') return null;
    const time = Date.parse(r.ts);
    if (Number.isNaN(time)) return null;
    return { time, ip: r.ip, method: r.method, path: normalisePath(r.path), status: r.status, ms: r.ms ?? 0 };
  } catch {
    return null;
  }
}
```

### Aggregation

```js
// src/stats.js
export class Stats {
  total = 0;
  malformed = 0;
  statusClasses = { '2xx': 0, '3xx': 0, '4xx': 0, '5xx': 0 };
  paths = new Map();
  ips = new Map();
  perMinute = new Map();
  // Latency histogram in 10 ms buckets (0–10s) → constant memory, approximate percentiles
  latency = new Uint32Array(1001);

  add(r) {
    this.total++;
    this.statusClasses[`${Math.floor(r.status / 100)}xx`] ??= 0;
    this.statusClasses[`${Math.floor(r.status / 100)}xx`]++;
    this.paths.set(r.path, (this.paths.get(r.path) ?? 0) + 1);
    this.ips.set(r.ip, (this.ips.get(r.ip) ?? 0) + 1);
    const minute = Math.floor(r.time / 60_000) * 60_000;
    this.perMinute.set(minute, (this.perMinute.get(minute) ?? 0) + 1);
    this.latency[Math.min(Math.floor(r.ms / 10), 1000)]++;
  }

  percentile(p) {
    const target = Math.ceil((p / 100) * this.total);
    let seen = 0;
    for (let i = 0; i < this.latency.length; i++) {
      seen += this.latency[i];
      if (seen >= target) return i * 10;
    }
    return 10_000;
  }

  static top(map, n) {
    return [...map].sort((a, b) => b[1] - a[1]).slice(0, n).map(([key, count]) => ({ key, count }));
  }

  summary() {
    const busiest = Stats.top(this.perMinute, 1)[0];
    return {
      total: this.total,
      malformed: this.malformed,
      statusClasses: this.statusClasses,
      errorRate: this.total ? +(this.statusClasses['5xx'] / this.total * 100).toFixed(2) : 0,
      latencyMs: { p50: this.percentile(50), p95: this.percentile(95), p99: this.percentile(99) },
      topPaths: Stats.top(this.paths, 10),
      topIps: Stats.top(this.ips, 10),
      busiestMinute: busiest ? { minute: new Date(busiest.key).toISOString(), requests: busiest.count } : null,
    };
  }
}
```

Note: `paths` and `ips` maps grow with **distinct** keys, not with lines — path normalisation keeps them small.

### Report

```js
// src/report.js
export function printText(s) {
  console.log(`Requests: ${s.total.toLocaleString()} (malformed lines skipped: ${s.malformed})`);
  console.log(`Status:   ${Object.entries(s.statusClasses).map(([k, v]) => `${k}=${v}`).join('  ')}`);
  console.log(`5xx rate: ${s.errorRate}%`);
  console.log(`Latency:  p50=${s.latencyMs.p50}ms p95=${s.latencyMs.p95}ms p99=${s.latencyMs.p99}ms`);
  if (s.busiestMinute) console.log(`Busiest:  ${s.busiestMinute.minute} (${s.busiestMinute.requests} req)`);
  console.log('\nTop paths:');
  console.table(s.topPaths);
  console.log('Top IPs:');
  console.table(s.topIps);
}
```

### CLI

```js
#!/usr/bin/env node
// bin/loganalyze.js
import { parseArgs } from 'node:util';
import { lines } from '../src/source.js';
import { parseLine } from '../src/parse.js';
import { Stats } from '../src/stats.js';
import { printText } from '../src/report.js';

const { values, positionals } = parseArgs({
  options: { since: { type: 'string' }, until: { type: 'string' }, json: { type: 'boolean' } },
  allowPositionals: true,
});
if (positionals.length === 0) {
  console.error('usage: loganalyze <file|-> [...] [--since ISO] [--until ISO] [--json]');
  process.exit(2);
}

const since = values.since ? Date.parse(values.since) : -Infinity;
const until = values.until ? Date.parse(values.until) : Infinity;
const stats = new Stats();
const started = performance.now();
let read = 0;

const progress = setInterval(() => {
  const rate = Math.round(read / ((performance.now() - started) / 1000));
  process.stderr.write(`\r${read.toLocaleString()} lines (${rate.toLocaleString()}/s)`);
}, 500);

try {
  for (const file of positionals) {
    for await (const line of lines(file)) {
      read++;
      const record = parseLine(line);
      if (!record) { stats.malformed++; continue; }
      if (record.time < since || record.time > until) continue;
      stats.add(record);
    }
  }
} catch (err) {
  clearInterval(progress);
  console.error(`\nfailed: ${err.message}`);
  process.exit(1);
}

clearInterval(progress);
process.stderr.write(`\rprocessed ${read.toLocaleString()} lines in ${((performance.now() - started) / 1000).toFixed(1)}s\n`);
const summary = stats.summary();
values.json ? console.log(JSON.stringify(summary, null, 2)) : printText(summary);
```
