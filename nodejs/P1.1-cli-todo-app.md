# P1.1 — CLI Todo App

**Phase:** 1 — Node.js Runtime Fundamentals
**Topic:** Subcommands, `parseArgs`, JSON file persistence, exit codes, coloured output
**Type:** Project

---

## Goal

Build a `todo` command-line tool that stores tasks in a JSON file in the user's home directory. Practice argument parsing, file persistence, correct use of stdout/stderr, and exit codes.

## What to Build

- `todo add "Buy milk" [--priority high|medium|low] [--due 2026-11-01]`
- `todo list [--all] [--priority high]` — pending tasks by default, `--all` includes done
- `todo done <id>` — mark a task complete
- `todo remove <id>` — delete a task
- `todo clear --done` — remove all completed tasks
- `todo --help` and `todo <command> --help`
- Data stored in `~/.todo.json` (override with `TODO_FILE` env var)
- Coloured output (`picocolors`), disabled automatically when stdout isn't a TTY
- Exit code `0` on success, `1` on runtime errors, `2` on usage errors

## Architecture

```
todo-cli/
├── package.json
├── bin/
│   └── todo.js          ← entry point with shebang
└── src/
    ├── cli.js           ← argument parsing & command dispatch
    ├── commands.js      ← add / list / done / remove / clear
    ├── store.js         ← load & save the JSON file
    ├── format.js        ← table output with colours
    └── errors.js        ← UsageError class
```

### Store

```js
// src/store.js
import { readFile, writeFile, rename } from 'node:fs/promises';
import { homedir } from 'node:os';
import { join } from 'node:path';

export const FILE = process.env.TODO_FILE ?? join(homedir(), '.todo.json');

export async function load() {
  try {
    return JSON.parse(await readFile(FILE, 'utf8'));
  } catch (err) {
    if (err.code === 'ENOENT') return { nextId: 1, tasks: [] };
    throw new Error(`could not read ${FILE}: ${err.message}`);
  }
}

export async function save(data) {
  // Write to a temp file then rename: a crash never leaves a half-written file
  const tmp = `${FILE}.${process.pid}.tmp`;
  await writeFile(tmp, JSON.stringify(data, null, 2));
  await rename(tmp, FILE);
}
```

### Commands

```js
// src/commands.js
import { load, save } from './store.js';
import { UsageError } from './errors.js';
import { formatTasks } from './format.js';

const PRIORITIES = ['high', 'medium', 'low'];

function parseId(raw) {
  const id = Number.parseInt(raw, 10);
  if (!Number.isInteger(id)) throw new UsageError(`invalid id "${raw}"`);
  return id;
}

export async function add([title], { priority = 'medium', due }) {
  if (!title) throw new UsageError('add requires a title');
  if (!PRIORITIES.includes(priority)) {
    throw new UsageError(`priority must be one of ${PRIORITIES.join(', ')}`);
  }
  if (due && Number.isNaN(Date.parse(due))) throw new UsageError(`invalid date "${due}"`);

  const data = await load();
  const task = { id: data.nextId++, title, priority, due: due ?? null, done: false, createdAt: new Date().toISOString() };
  data.tasks.push(task);
  await save(data);
  console.log(`added #${task.id}: ${task.title}`);
}

export async function list(_args, { all = false, priority }) {
  const { tasks } = await load();
  const shown = tasks
    .filter(t => all || !t.done)
    .filter(t => !priority || t.priority === priority);
  console.log(shown.length ? formatTasks(shown) : 'nothing to do 🎉');
}

export async function done([rawId]) {
  const id = parseId(rawId);
  const data = await load();
  const task = data.tasks.find(t => t.id === id);
  if (!task) throw new Error(`no task with id ${id}`);
  task.done = true;
  await save(data);
  console.log(`completed #${id}`);
}

export async function remove([rawId]) {
  const id = parseId(rawId);
  const data = await load();
  const before = data.tasks.length;
  data.tasks = data.tasks.filter(t => t.id !== id);
  if (data.tasks.length === before) throw new Error(`no task with id ${id}`);
  await save(data);
  console.log(`removed #${id}`);
}

export async function clear(_args, { done: onlyDone }) {
  if (!onlyDone) throw new UsageError('use `todo clear --done` to remove completed tasks');
  const data = await load();
  const removed = data.tasks.filter(t => t.done).length;
  data.tasks = data.tasks.filter(t => !t.done);
  await save(data);
  console.log(`cleared ${removed} completed task(s)`);
}
```

### Formatting

```js
// src/format.js
import pc from 'picocolors';

const colour = { high: pc.red, medium: pc.yellow, low: pc.dim };

export function formatTasks(tasks) {
  return tasks
    .map(t => {
      const box = t.done ? pc.green('✔') : '☐';
      const pri = colour[t.priority](t.priority.padEnd(6));
      const due = t.due ? pc.cyan(` (due ${t.due})`) : '';
      const title = t.done ? pc.strikethrough(t.title) : t.title;
      return `${box} ${String(t.id).padStart(3)}  ${pri} ${title}${due}`;
    })
    .join('\n');
}
```

`picocolors` already disables colours when output is not a TTY or `NO_COLOR` is set.

### CLI dispatch

```js
// src/cli.js
import { parseArgs } from 'node:util';
import * as commands from './commands.js';
import { UsageError } from './errors.js';

const HELP = `Usage: todo <command> [options]

Commands:
  add <title> [--priority high|medium|low] [--due YYYY-MM-DD]
  list [--all] [--priority P]
  done <id>
  remove <id>
  clear --done`;

const OPTIONS = {
  priority: { type: 'string', short: 'p' },
  due:      { type: 'string' },
  all:      { type: 'boolean', short: 'a' },
  done:     { type: 'boolean' },
  help:     { type: 'boolean', short: 'h' },
};

export async function run(argv) {
  let parsed;
  try {
    parsed = parseArgs({ args: argv, options: OPTIONS, allowPositionals: true });
  } catch (err) {
    throw new UsageError(err.message);
  }
  const [command, ...rest] = parsed.positionals;

  if (parsed.values.help || !command) {
    console.log(HELP);
    return;
  }
  const handler = commands[command];
  if (typeof handler !== 'function') throw new UsageError(`unknown command "${command}"\n\n${HELP}`);
  await handler(rest, parsed.values);
}
```
