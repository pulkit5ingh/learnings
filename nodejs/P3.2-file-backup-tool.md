# P3.2 — File Backup Tool

**Phase:** 3 — File System & Streams
**Topic:** Directory walking, streaming hashes, content-addressed storage, gzip, manifests, restore, progress
**Type:** Project

---

## Goal

Build `snap` — an incremental, deduplicating backup tool. Each backup stores a **manifest** of every file's path, size, mode, and SHA-256 hash. File contents are stored once in a content-addressed object store (`objects/ab/cdef…gz`), so unchanged files and duplicate files cost nothing in later backups. Everything streams, so files of any size work.

## What to Build

- `snap backup <source> --repo <dir>` — create a snapshot
- `snap list --repo <dir>` — list snapshots with date, file count, total and new bytes
- `snap restore <snapshot-id> <target> --repo <dir>` — restore a snapshot
- `snap verify --repo <dir>` — re-hash every object and report corruption
- Incremental: files whose hash already exists in the store are not copied again
- Objects are gzip-compressed; hashes are computed on the **original** content
- Ignore patterns via `--ignore node_modules,.git`
- Progress on stderr: files scanned, bytes stored, bytes skipped
- Restored files get their original permissions and modification times
- Writes are atomic: an interrupted backup never leaves a half-written object or manifest

## Implementation

### Walking

```js
// src/walk.js
import { readdir } from 'node:fs/promises';
import { join } from 'node:path';

export async function* walk(dir, ignore, base = dir) {
  for (const entry of await readdir(dir, { withFileTypes: true })) {
    if (ignore.has(entry.name)) continue;
    const full = join(dir, entry.name);
    if (entry.isDirectory()) yield* walk(full, ignore, base);
    else if (entry.isFile()) yield full;
    // symlinks, sockets, etc. are skipped deliberately
  }
}
```

### Hashing a stream

```js
// src/hash.js
import { createReadStream } from 'node:fs';
import { createHash } from 'node:crypto';
import { pipeline } from 'node:stream/promises';

export async function hashFile(path) {
  const hash = createHash('sha256');
  await pipeline(createReadStream(path), hash);
  return hash.digest('hex');
}
```

A `Hash` object is a Transform stream, so it can sit at the end of a pipeline.

### Object store

```js
// src/store.js
import { createReadStream, createWriteStream } from 'node:fs';
import { mkdir, rename, access, rm } from 'node:fs/promises';
import { createGzip, createGunzip } from 'node:zlib';
import { pipeline } from 'node:stream/promises';
import { join, dirname } from 'node:path';
import { randomUUID } from 'node:crypto';

export class ObjectStore {
  constructor(repo) { this.root = join(repo, 'objects'); }

  pathFor(hash) { return join(this.root, hash.slice(0, 2), `${hash}.gz`); }

  async has(hash) {
    try { await access(this.pathFor(hash)); return true; }
    catch { return false; }
  }

  async put(hash, sourcePath) {
    const dest = this.pathFor(hash);
    await mkdir(dirname(dest), { recursive: true });
    const tmp = `${dest}.${randomUUID()}.tmp`;
    try {
      await pipeline(createReadStream(sourcePath), createGzip(), createWriteStream(tmp));
      await rename(tmp, dest);     // atomic: an object is either complete or absent
    } catch (err) {
      await rm(tmp, { force: true });
      throw err;
    }
  }

  read(hash) {
    const file = createReadStream(this.pathFor(hash));
    const gunzip = createGunzip();
    file.on('error', (err) => gunzip.destroy(err));   // .pipe() doesn't forward errors
    return file.pipe(gunzip);
  }
}
```

Using `access` here is acceptable: a stale "missing" answer only causes a redundant write of identical content.

### Backup

```js
// src/backup.js
import { stat, mkdir, writeFile, rename } from 'node:fs/promises';
import { join, relative, resolve } from 'node:path';
import { walk } from './walk.js';
import { hashFile } from './hash.js';
import { ObjectStore } from './store.js';

export async function backup(source, repo, { ignore, onProgress }) {
  const store = new ObjectStore(repo);
  const root = resolve(source);
  const files = [];
  const totals = { files: 0, bytes: 0, newBytes: 0, skippedBytes: 0 };

  for await (const full of walk(root, ignore)) {
    const s = await stat(full);
    const hash = await hashFile(full);
    if (await store.has(hash)) {
      totals.skippedBytes += s.size;
    } else {
      await store.put(hash, full);
      totals.newBytes += s.size;
    }
    totals.files++;
    totals.bytes += s.size;
    files.push({ path: relative(root, full), size: s.size, mode: s.mode & 0o777, mtimeMs: s.mtimeMs, hash });
    onProgress?.(totals);
  }

  const id = new Date().toISOString().replaceAll(':', '-').replace('.', '-');
  const manifest = { id, source: root, createdAt: new Date().toISOString(), totals, files };
  const dir = join(repo, 'snapshots');
  await mkdir(dir, { recursive: true });
  const tmp = join(dir, `.${id}.tmp`);
  await writeFile(tmp, JSON.stringify(manifest, null, 2));
  await rename(tmp, join(dir, `${id}.json`));
  return manifest;
}
```

### Restore

```js
// src/restore.js
import { readFile, mkdir, chmod, utimes } from 'node:fs/promises';
import { createWriteStream } from 'node:fs';
import { pipeline } from 'node:stream/promises';
import { join, dirname, resolve, sep } from 'node:path';
import { ObjectStore } from './store.js';

export async function restore(repo, id, target, { onFile } = {}) {
  const manifest = JSON.parse(await readFile(join(repo, 'snapshots', `${id}.json`), 'utf8'));
  const store = new ObjectStore(repo);
  const root = resolve(target);

  for (const f of manifest.files) {
    const dest = resolve(root, f.path);
    // Never write outside the target, even if a manifest was tampered with
    if (dest !== root && !dest.startsWith(root + sep)) throw new Error(`unsafe path in manifest: ${f.path}`);
    await mkdir(dirname(dest), { recursive: true });
    await pipeline(store.read(f.hash), createWriteStream(dest));
    await chmod(dest, f.mode);
    const mtime = new Date(f.mtimeMs);
    await utimes(dest, mtime, mtime);
    onFile?.(f);
  }
  return manifest;
}
```

### Verify

```js
// src/verify.js
import { readdir } from 'node:fs/promises';
import { createHash } from 'node:crypto';
import { pipeline } from 'node:stream/promises';
import { join, basename } from 'node:path';
import { ObjectStore } from './store.js';

export async function verify(repo) {
  const store = new ObjectStore(repo);
  const bad = [];
  let checked = 0;
  for (const prefix of await readdir(store.root)) {
    for (const file of await readdir(join(store.root, prefix))) {
      if (!file.endsWith('.gz')) continue;
      const expected = basename(file, '.gz');
      const hash = createHash('sha256');
      try {
        await pipeline(store.read(expected), hash);
        if (hash.digest('hex') !== expected) bad.push({ hash: expected, reason: 'hash mismatch' });
      } catch (err) {
        bad.push({ hash: expected, reason: err.code ?? err.message });
      }
      checked++;
    }
  }
  return { checked, bad };
}
```

### CLI

```js
#!/usr/bin/env node
// bin/snap.js
import { parseArgs } from 'node:util';
import { readdir, readFile } from 'node:fs/promises';
import { join } from 'node:path';
import { backup } from '../src/backup.js';
import { restore } from '../src/restore.js';
import { verify } from '../src/verify.js';

const { values, positionals } = parseArgs({
  options: {
    repo:   { type: 'string', default: './backups' },
    ignore: { type: 'string', default: 'node_modules,.git' },
  },
  allowPositionals: true,
});
const [cmd, ...args] = positionals;
const mb = (n) => `${(n / 1024 / 1024).toFixed(1)} MB`;

try {
  if (cmd === 'backup' && args[0]) {
    const m = await backup(args[0], values.repo, {
      ignore: new Set(values.ignore.split(',').filter(Boolean)),
      onProgress: (t) => process.stderr.write(`\r${t.files} files, new ${mb(t.newBytes)}, unchanged ${mb(t.skippedBytes)}`),
    });
    process.stderr.write('\n');
    console.log(`snapshot ${m.id}: ${m.totals.files} files, ${mb(m.totals.bytes)} total, ${mb(m.totals.newBytes)} new`);
  } else if (cmd === 'list') {
    const dir = join(values.repo, 'snapshots');
    for (const f of (await readdir(dir)).filter(f => f.endsWith('.json')).sort()) {
      const m = JSON.parse(await readFile(join(dir, f), 'utf8'));
      console.log(`${m.id}  ${String(m.totals.files).padStart(6)} files  ${mb(m.totals.bytes).padStart(10)}  +${mb(m.totals.newBytes)}`);
    }
  } else if (cmd === 'restore' && args.length === 2) {
    let n = 0;
    await restore(values.repo, args[0], args[1], { onFile: () => process.stderr.write(`\rrestored ${++n} files`) });
    process.stderr.write('\n');
  } else if (cmd === 'verify') {
    const { checked, bad } = await verify(values.repo);
    console.log(`checked ${checked} objects, ${bad.length} problems`);
    bad.forEach(b => console.log(`  ${b.hash}: ${b.reason}`));
    if (bad.length) process.exitCode = 1;
  } else {
    console.error('usage: snap backup <src> | list | restore <id> <target> | verify   [--repo DIR] [--ignore a,b]');
    process.exitCode = 2;
  }
} catch (err) {
  console.error(`snap: ${err.message}`);
  process.exitCode = 1;
}
```

### Try it

```bash
node bin/snap.js backup ~/projects/some-app --repo /tmp/snaps
node bin/snap.js backup ~/projects/some-app --repo /tmp/snaps     # second run: ~0 MB new
touch ~/projects/some-app/README.md && node bin/snap.js backup ~/projects/some-app --repo /tmp/snaps
node bin/snap.js list --repo /tmp/snaps
node bin/snap.js restore <id> /tmp/restored --repo /tmp/snaps
diff -r ~/projects/some-app /tmp/restored -x node_modules -x .git && echo identical

# Corrupt one object in place, then verify
obj=$(find /tmp/snaps/objects -name '*.gz' | head -1)
printf 'XXXX' | dd of="$obj" bs=1 seek=20 conv=notrunc
node bin/snap.js verify --repo /tmp/snaps; echo "exit: $?"
```

## Deliverables

- [ ] A second backup of an unchanged directory stores ~0 new bytes
- [ ] Duplicate files inside one backup are stored once
- [ ] Hashing, compression, and restore all stream — large files don't increase memory
- [ ] Objects and manifests are written to temp files and renamed atomically
- [ ] `restore` reproduces the source exactly (`diff -r` is clean), including permissions and mtimes
- [ ] `restore` refuses manifest paths that escape the target directory
- [ ] `verify` detects a corrupted object and exits with code 1
- [ ] Ignore patterns skip `node_modules` and `.git`
