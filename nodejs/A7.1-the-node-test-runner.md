## Goal
Test Node code with the runner that ships with Node — no Jest, no dependencies — and set up a fast feedback loop with watch mode and coverage.

## Background

### A complete runner, built in
`node:test` is stable since Node 20. It provides `test`/`describe`/`it`, hooks (`before`, `after`, `beforeEach`, `afterEach`), mocking (A7.2), reporters, watch mode, filtering, and coverage. Pair it with `node:assert/strict`.

### Running tests
`node --test` with no arguments finds files named like `*.test.js`, `*-test.js`, `*_test.js`, or anything in a `test/` directory, and runs each file in its **own process**, in parallel. You can also pass globs: `node --test 'src/**/*.test.js'`.

### Assertions
Always use the strict variant: `import assert from 'node:assert/strict'`.

| Assertion | Checks |
|-----------|--------|
| `assert.equal(a, b)` | `a === b` |
| `assert.deepEqual(a, b)` | Structural equality, strict types |
| `assert.ok(v)` | Truthy |
| `assert.match(str, /re/)` | Regex match |
| `assert.throws(fn, expected)` | Sync function throws |
| `await assert.rejects(promise, expected)` | Promise rejects |

### Async tests
An `async` test passes when its promise resolves. Forgetting `await` on `assert.rejects` is a classic bug: the test finishes before the assertion runs.

### Structure and focus
Group with `describe`; nest subtests with `await t.test()`. Use `test.skip`, `test.todo`, `{ only: true }` with `--test-only`, and `--test-name-pattern` to filter.

### Coverage
`node --test --experimental-test-coverage` reports line, branch, and function coverage. Thresholds such as `--test-coverage-lines=90` fail the run below the limit (Node 22.8+).

```js
// src/cart.js
export class Cart {
  #items = new Map();

  add(sku, price, qty = 1) {
    if (!Number.isInteger(qty) || qty < 1) throw new RangeError('qty must be a positive integer');
    if (!(price >= 0)) throw new TypeError('price must be a non-negative number');
    const existing = this.#items.get(sku);
    this.#items.set(sku, { sku, price, qty: (existing?.qty ?? 0) + qty });
    return this;
  }

  remove(sku) { return this.#items.delete(sku); }
  get items() { return [...this.#items.values()]; }

  total({ discountPercent = 0 } = {}) {
    const subtotal = this.items.reduce((sum, i) => sum + i.price * i.qty, 0);
    return Math.round(subtotal * (1 - discountPercent / 100) * 100) / 100;
  }
}

export async function loadCart(fetchItems) {
  const items = await fetchItems();
  if (!Array.isArray(items)) throw new Error('invalid cart payload');
  const cart = new Cart();
  for (const { sku, price, qty } of items) cart.add(sku, price, qty);
  return cart;
}
```

### 2. First tests

```js
// test/cart.test.js
import { describe, it, beforeEach } from 'node:test';
import assert from 'node:assert/strict';
import { Cart, loadCart } from '../src/cart.js';

describe('Cart', () => {
  let cart;
  beforeEach(() => { cart = new Cart(); });

  it('starts empty', () => {
    assert.deepEqual(cart.items, []);
    assert.equal(cart.total(), 0);
  });

  it('merges quantities for the same sku', () => {
    cart.add('apple', 0.5, 2).add('apple', 0.5, 3);
    assert.deepEqual(cart.items, [{ sku: 'apple', price: 0.5, qty: 5 }]);
  });

  it('computes totals with discounts, rounded to cents', () => {
    cart.add('a', 19.99, 3);
    assert.equal(cart.total(), 59.97);
    assert.equal(cart.total({ discountPercent: 10 }), 53.97);
  });

  it('rejects invalid input', () => {
    assert.throws(() => cart.add('x', 1, 0), RangeError);
    assert.throws(() => cart.add('x', 1, 1.5), { name: 'RangeError', message: /positive integer/ });
    assert.throws(() => cart.add('x', NaN), TypeError);
  });
});

describe('loadCart', () => {
  it('builds a cart from fetched items', async () => {
    const cart = await loadCart(async () => [{ sku: 'a', price: 2, qty: 2 }]);
    assert.equal(cart.total(), 4);
  });

  it('rejects invalid payloads and propagates errors', async () => {
    await assert.rejects(loadCart(async () => ({})), /invalid cart payload/);
    await assert.rejects(loadCart(async () => { throw new Error('network down'); }), { message: 'network down' });
  });
});
```

```js
// test/structure.test.js
import { test } from 'node:test';
import assert from 'node:assert/strict';
import { Cart } from '../src/cart.js';

test('discount table', async (t) => {
  for (const [pct, expected] of [[0, 100], [10, 90], [25, 75], [100, 0]]) {
    await t.test(`${pct}% off`, () => {
      assert.equal(new Cart().add('x', 100).total({ discountPercent: pct }), expected);
    });
  }
});

test.skip('payment gateway integration', () => {});
test.todo('supports coupon codes');
test('focused test', { only: true }, () => assert.ok(true));
```

### 5. Hooks with real resources

```js
// test/server.test.js
import { describe, it, before, after } from 'node:test';
import assert from 'node:assert/strict';
import { createServer } from 'node:http';
import { once } from 'node:events';

describe('HTTP server', () => {
  let server, base;

  before(async () => {
    server = createServer((req, res) => {
      if (req.url === '/health') return res.end(JSON.stringify({ ok: true }));
      res.writeHead(404).end();
    }).listen(0);                               // random free port
    await once(server, 'listening');
    base = `http://localhost:${server.address().port}`;
  });

  after(() => server.close());

  it('GET /health returns ok', async () => {
    const res = await fetch(`${base}/health`);
    assert.equal(res.status, 200);
    assert.deepEqual(await res.json(), { ok: true });
  });

  it('unknown routes return 404', async () => {
    assert.equal((await fetch(`${base}/nope`)).status, 404);
  });
});
```
