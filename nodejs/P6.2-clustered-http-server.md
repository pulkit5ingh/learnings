# P6.2 — Clustered HTTP Server

**Phase:** 6 — Performance & Concurrency
**Topic:** Cluster supervisor, health reporting over IPC, crash recovery, zero-downtime reloads, aggregated metrics, offloading CPU work
**Type:** Project

---

## Goal

Build a production-style process supervisor for an HTTP API: a primary that runs one worker per core, restarts crashed or unhealthy workers, performs rolling reloads with zero dropped requests, and exposes aggregated metrics from all workers — the core of what PM2 does, in ~200 lines.

## What to Build

- `node supervisor.js` starts `availableParallelism()` workers (override with `WORKERS`)
- Workers run `app.js` — an API with `/`, `/work` (CPU-heavy, offloaded to a worker thread), `/slow`, and `/crash`
- Workers report **heartbeats** to the primary every second with memory, event loop delay, and request counts
- The primary restarts a worker that:
  - exits unexpectedly (with exponential backoff and crash-loop protection)
  - misses 3 heartbeats (hung event loop)
  - exceeds a memory limit (`MAX_RSS_MB`)
- `SIGHUP` → rolling reload (one worker at a time, new worker must be listening before the old one drains)
- `SIGTERM`/`SIGINT` → graceful shutdown of all workers with a deadline
- `GET /_status` on any worker returns cluster-wide stats aggregated by the primary
- Structured JSON logs including `pid` and `workerId`
- A load test proving zero errors during a reload

## Architecture

```
cluster-server/
├── package.json
├── supervisor.js        ← primary: fork, monitor, reload, shutdown
├── app.js               ← worker: HTTP server + heartbeats + graceful close
├── cpu-task.js          ← worker_threads task for /work
└── lib/
    ├── log.js
    └── ipc.js           ← request/response helper over process.send
```

import cluster from 'node:cluster';

export function log(level, msg, extra = {}) {
  const line = { ts: new Date().toISOString(), level, pid: process.pid, role: cluster.isPrimary ? 'primary' : 'worker', workerId: cluster.worker?.id, msg, ...extra };
  (level === 'error' ? process.stderr : process.stdout).write(JSON.stringify(line) + '\n');
}

### IPC request/response helper (worker side)

```js
// lib/ipc.js
let nextId = 0;
const pending = new Map();

process.on('message', (msg) => {
  if (msg?.type === 'reply' && pending.has(msg.id)) {
    pending.get(msg.id)(msg.data);
    pending.delete(msg.id);
  }
});

export function askPrimary(type, timeoutMs = 1000) {
  return new Promise((resolve, reject) => {
    const id = ++nextId;
    const timer = setTimeout(() => { pending.delete(id); reject(new Error('primary did not reply')); }, timeoutMs);
    pending.set(id, (data) => { clearTimeout(timer); resolve(data); });
    process.send({ type, id });
  });
}
```

### Worker app

```js
// app.js
import { createServer } from 'node:http';
import { Worker } from 'node:worker_threads';
import { monitorEventLoopDelay } from 'node:perf_hooks';
import { setTimeout as sleep } from 'node:timers/promises';
import { log } from './lib/log.js';
import { askPrimary } from './lib/ipc.js';

const stats = { requests: 0, errors: 0, inFlight: 0 };
const loop = monitorEventLoopDelay({ resolution: 20 });
loop.enable();

const runCpuTask = (rounds) => new Promise((resolve, reject) => {
  const w = new Worker(new URL('./cpu-task.js', import.meta.url), { workerData: { seed: String(Math.random()), rounds } });
  w.once('message', resolve).once('error', reject);
});

const json = (res, status, body) => { res.writeHead(status, { 'content-type': 'application/json' }); res.end(JSON.stringify(body)); };

const server = createServer(async (req, res) => {
  stats.requests++; stats.inFlight++;
  res.on('close', () => stats.inFlight--);
  try {
    switch (new URL(req.url, 'http://x').pathname) {
      case '/': return json(res, 200, { ok: true, pid: process.pid });
      case '/work': return json(res, 200, { hash: (await runCpuTask(200_000)).slice(0, 16), pid: process.pid });
      case '/slow': await sleep(2000); return json(res, 200, { slow: true, pid: process.pid });
      case '/crash': setImmediate(() => { throw new Error('simulated crash'); }); return json(res, 202, { crashing: process.pid });
      case '/hang': { const end = Date.now() + 10_000; while (Date.now() < end); return json(res, 200, {}); }   // blocks heartbeats
      case '/_status': return json(res, 200, await askPrimary('status'));
      default: return json(res, 404, { error: 'not found' });
    }
  } catch (err) {
    stats.errors++;
    log('error', 'request failed', { err: err.message });
    if (!res.headersSent) json(res, 500, { error: 'internal error' });
  }
});
server.keepAliveTimeout = 5000;
server.listen(Number(process.env.PORT ?? 3000));

// Heartbeats: the primary treats silence as "hung"
setInterval(() => {
  process.send?.({
    type: 'heartbeat',
    data: { ...stats, rssMb: Math.round(process.memoryUsage().rss / 1e6), loopP99Ms: +(loop.percentile(99) / 1e6).toFixed(1) },
  });
  loop.reset();
}, 1000).unref();

// Graceful shutdown requested by the primary
process.on('message', (msg) => {
  if (msg?.type !== 'shutdown') return;
  log('info', 'draining', { inFlight: stats.inFlight });
  server.close(() => process.exit(0));
  server.closeIdleConnections();
  setTimeout(() => { log('error', 'drain timeout, forcing exit'); process.exit(1); }, msg.deadlineMs ?? 10_000).unref();
});
```

### Supervisor

```js
// supervisor.js
import cluster from 'node:cluster';
import { availableParallelism } from 'node:os';
import { once } from 'node:events';
import { join } from 'node:path';
import { log } from './lib/log.js';

const WORKERS = Number(process.env.WORKERS ?? availableParallelism());
const MAX_RSS_MB = Number(process.env.MAX_RSS_MB ?? 512);
const HEARTBEAT_LIMIT_MS = 3500;

cluster.setupPrimary({ exec: join(import.meta.dirname, 'app.js') });

const health = new Map();         // worker.id → { lastSeen, stats }
const crashes = [];
let shuttingDown = false, reloading = false;

function fork() {
  const worker = cluster.fork();
  health.set(worker.id, { lastSeen: Date.now(), stats: null });

  worker.on('message', (msg) => {
    if (msg.type === 'heartbeat') health.set(worker.id, { lastSeen: Date.now(), stats: msg.data });
    if (msg.type === 'status') worker.send({ type: 'reply', id: msg.id, data: clusterStatus() });
  });
  return worker;
}

function clusterStatus() {
  const workers = Object.values(cluster.workers).map((w) => ({ id: w.id, pid: w.process.pid, ...health.get(w.id)?.stats }));
  const sum = (k) => workers.reduce((n, w) => n + (w[k] ?? 0), 0);
  return { primary: process.pid, workers, totals: { requests: sum('requests'), errors: sum('errors'), inFlight: sum('inFlight'), rssMb: sum('rssMb') } };
}

async function stop(worker, reason, deadlineMs = 10_000) {
  if (!worker.isConnected()) return;
  log('warn', 'stopping worker', { id: worker.id, pid: worker.process.pid, reason });
  worker.send({ type: 'shutdown', deadlineMs });
  const kill = setTimeout(() => worker.process.kill('SIGKILL'), deadlineMs + 2000);
  await once(worker, 'exit');
  clearTimeout(kill);
}

// Unexpected exits → restart with backoff and crash-loop protection
cluster.on('exit', (worker, code, signal) => {
  health.delete(worker.id);
  if (shuttingDown || worker.replaced) return;
  const now = Date.now();
  crashes.push(now);
  while (crashes[0] < now - 60_000) crashes.shift();
  log('error', 'worker exited unexpectedly', { pid: worker.process.pid, code, signal, crashesLastMinute: crashes.length });
  if (crashes.length > WORKERS * 5) { log('error', 'crash loop detected, exiting'); process.exit(1); }
  const delay = Math.min(100 * 2 ** (crashes.length - 1), 10_000);
  setTimeout(fork, delay);
});

// Watchdog: hung or bloated workers get replaced
setInterval(() => {
  if (reloading || shuttingDown) return;
  for (const worker of Object.values(cluster.workers)) {
    const h = health.get(worker.id);
    if (!h || worker.replaced) continue;          // already being replaced
    if (Date.now() - h.lastSeen > HEARTBEAT_LIMIT_MS) {
      log('error', 'worker missed heartbeats, killing', { pid: worker.process.pid });
      worker.process.kill('SIGKILL');             // it can't drain — its loop is blocked
    } else if (h.stats?.rssMb > MAX_RSS_MB) {
      replace(worker, `rss ${h.stats.rssMb}MB > ${MAX_RSS_MB}MB`);
    }
  }
}, 1000).unref();

async function replace(worker, reason) {
  worker.replaced = true;
  const fresh = fork();
  await once(fresh, 'listening');
  await stop(worker, reason);
}

async function rollingReload() {
  if (reloading) return;
  reloading = true;
  log('info', 'rolling reload started');
  for (const worker of Object.values(cluster.workers)) await replace(worker, 'reload');
  reloading = false;
  log('info', 'rolling reload complete', { workers: Object.keys(cluster.workers).length });
}

async function shutdown(signal) {
  if (shuttingDown) return;
  shuttingDown = true;
  log('info', 'shutting down', { signal });
  await Promise.all(Object.values(cluster.workers).map((w) => stop(w, signal)));
  log('info', 'all workers stopped');
  process.exit(0);
}

for (let i = 0; i < WORKERS; i++) fork();
process.on('SIGHUP', rollingReload);
process.on('SIGTERM', () => shutdown('SIGTERM'));
process.on('SIGINT', () => shutdown('SIGINT'));
log('info', 'supervisor started', { workers: WORKERS, reload: `kill -HUP ${process.pid}` });
```

### Try it

```bash
node supervisor.js | npx pino-pretty        # or just read the JSON lines
curl -s localhost:3000/_status | jq .totals

# Crash recovery
curl localhost:3000/crash && sleep 1 && curl -s localhost:3000/_status | jq '.workers | length'

# Hung worker detection
curl --max-time 2 localhost:3000/hang; sleep 5; curl -s localhost:3000/_status | jq '.workers[].pid'

# Zero-downtime reload under load
npx autocannon -c 50 -d 20 localhost:3000/slow &
sleep 4 && kill -HUP $(pgrep -f supervisor.js)
wait                                         # autocannon summary: 0 errors, 0 timeouts

# CPU offload keeps / fast
npx autocannon -c 20 -d 10 localhost:3000/work & sleep 1; time curl -s localhost:3000/
```

## Deliverables

- [ ] One worker per core serves requests on a shared port
- [ ] Crashed workers restart with exponential backoff; a crash loop stops the supervisor
- [ ] A worker with a blocked event loop is detected via missed heartbeats and replaced
- [ ] Workers exceeding `MAX_RSS_MB` are replaced gracefully
- [ ] `SIGHUP` performs a rolling reload with zero failed requests under load
- [ ] `SIGTERM` drains all workers before exiting, with a forced-kill deadline
- [ ] `/_status` returns aggregated stats from every worker via the primary
- [ ] `/work` runs in a worker thread and doesn't slow down `/`
- [ ] Logs are structured JSON with pid and worker id
