# P2.1 — Concurrent URL Health Checker

**Phase:** 2 — Asynchronous Programming
**Topic:** Concurrency pools, per-request timeouts, retries with backoff, `Promise.allSettled`, reporting
**Type:** Project

---

## Goal

Build `healthcheck` — a CLI that reads a list of URLs and checks them concurrently with a bounded pool, a per-request timeout, and retries with exponential backoff. It prints a summary table and exits non-zero if any URL is unhealthy, so it can run in CI or cron.

## What to Build

- Read URLs from a file (`healthcheck urls.txt`) or stdin (`cat urls.txt | healthcheck -`)
- `--concurrency N` (default 5) — never more than N requests in flight
- `--timeout MS` (default 5000) — abort each attempt after this long
- `--retries N` (default 2) — retry network errors, timeouts, and 5xx with backoff (200 ms, 400 ms, 800 ms… plus jitter)
- `--json` — machine-readable output instead of a table
- Per URL: status code, latency, attempts, healthy/unhealthy, error message
- Progress on stderr while running
- Exit code `0` if all healthy, `1` if any unhealthy, `2` on usage errors
- Ctrl+C aborts all in-flight requests and prints partial results

## Implementation

### Reading input

```js
// src/input.js
import { createReadStream } from 'node:fs';
import { createInterface } from 'node:readline';

export async function readUrls(source) {
  const input = source === '-' ? process.stdin : createReadStream(source);
  const rl = createInterface({ input, crlfDelay: Infinity });
  const urls = [];
  for await (const raw of rl) {
    const line = raw.trim();
    if (!line || line.startsWith('#')) continue;
    try {
      const url = new URL(line);
      if (!['http:', 'https:'].includes(url.protocol)) throw new Error('unsupported protocol');
      urls.push(url.href);
    } catch {
      console.error(`skipping invalid URL: ${line}`);
    }
  }
  return [...new Set(urls)];
}
```

### Concurrency pool

```js
// src/pool.js
export async function mapLimit(items, limit, fn) {
  const results = new Array(items.length);
  let next = 0;
  const worker = async () => {
    while (next < items.length) {
      const i = next++;
      results[i] = await fn(items[i], i);
    }
  };
  await Promise.all(Array.from({ length: Math.min(limit, items.length) }, worker));
  return results;
}
```

### Checking a URL

```js
// src/check.js
import { setTimeout as sleep } from 'node:timers/promises';

const isRetryable = (result) => result.error !== undefined || result.status >= 500;

async function attempt(url, timeoutMs, signal) {
  const started = performance.now();
  try {
    const res = await fetch(url, {
      method: 'GET',
      redirect: 'follow',
      signal: AbortSignal.any([signal, AbortSignal.timeout(timeoutMs)]),
    });
    await res.body?.cancel();          // we only need the status — don't download the body
    return { status: res.status, latencyMs: Math.round(performance.now() - started) };
  } catch (err) {
    if (signal.aborted) throw err;     // user pressed Ctrl+C — stop retrying
    const reason = err.name === 'TimeoutError' ? `timeout after ${timeoutMs}ms` : (err.cause?.code ?? err.message);
    return { error: reason, latencyMs: Math.round(performance.now() - started) };
  }
}

export async function checkUrl(url, { timeoutMs, retries, signal }) {
  let result;
  for (let n = 0; n <= retries; n++) {
    result = await attempt(url, timeoutMs, signal);
    result.attempts = n + 1;
    if (!isRetryable(result) || n === retries) break;
    const backoff = 200 * 2 ** n + Math.random() * 100;   // exponential + jitter
    await sleep(backoff, undefined, { signal });
  }
  result.url = url;
  result.healthy = result.error === undefined && result.status >= 200 && result.status < 400;
  return result;
}
```

### Reporting

```js
// src/report.js
export function printTable(results) {
  const rows = results.map(r => ({
    url: r.url,
    status: r.status ?? '-',
    'latency (ms)': r.latencyMs,
    attempts: r.attempts,
    result: r.healthy ? 'OK' : `FAIL${r.error ? ` (${r.error})` : ''}`,
  }));
  console.table(rows);
  const failed = results.filter(r => !r.healthy).length;
  console.log(`${results.length - failed}/${results.length} healthy`);
}

export function printJson(results) {
  console.log(JSON.stringify({ checkedAt: new Date().toISOString(), results }, null, 2));
}
```

### Entry point

```js
#!/usr/bin/env node
// bin/healthcheck.js
import { parseArgs } from 'node:util';
import { readUrls } from '../src/input.js';
import { mapLimit } from '../src/pool.js';
import { checkUrl } from '../src/check.js';
import { printTable, printJson } from '../src/report.js';

function int(name, raw, min) {
  const n = Number.parseInt(raw, 10);
  if (!Number.isInteger(n) || n < min) throw new Error(`--${name} must be an integer >= ${min}`);
  return n;
}

let opts, source;
try {
  const { values, positionals } = parseArgs({
    options: {
      concurrency: { type: 'string', short: 'c', default: '5' },
      timeout:     { type: 'string', short: 't', default: '5000' },
      retries:     { type: 'string', short: 'r', default: '2' },
      json:        { type: 'boolean' },
    },
    allowPositionals: true,
  });
  if (positionals.length !== 1) throw new Error('usage: healthcheck <file|-> [options]');
  source = positionals[0];
  opts = {
    concurrency: int('concurrency', values.concurrency, 1),
    timeoutMs: int('timeout', values.timeout, 1),
    retries: int('retries', values.retries, 0),
    json: values.json,
  };
} catch (err) {
  console.error(err.message);
  process.exit(2);
}

const urls = await readUrls(source);
if (urls.length === 0) {
  console.error('no valid URLs to check');
  process.exit(2);
}

const controller = new AbortController();
process.once('SIGINT', () => {
  console.error('\naborting…');
  controller.abort();
});

let done = 0;
const results = await mapLimit(urls, opts.concurrency, async (url) => {
  try {
    return await checkUrl(url, { ...opts, signal: controller.signal });
  } catch {
    return { url, healthy: false, error: 'aborted', attempts: 0, latencyMs: 0 };
  } finally {
    process.stderr.write(`\rchecked ${++done}/${urls.length}`);
  }
});
process.stderr.write('\n');

opts.json ? printJson(results) : printTable(results);
process.exitCode = results.every(r => r.healthy) ? 0 : 1;
```

## Deliverables

- [ ] URLs are read from a file or stdin; blanks, comments, duplicates, and invalid URLs are skipped
- [ ] No more than `--concurrency` requests are ever in flight
- [ ] Each attempt is aborted after `--timeout` using `AbortSignal.timeout`
- [ ] Network errors, timeouts, and 5xx are retried with exponential backoff and jitter; 4xx are not
- [ ] Response bodies are cancelled, not downloaded
- [ ] Progress goes to stderr, results to stdout; `--json` output pipes cleanly into `jq`
- [ ] Exit codes: 0 all healthy, 1 any unhealthy, 2 usage error
- [ ] Ctrl+C aborts in-flight requests and still prints results
