## Goal
Build a small router and middleware pipeline on top of `node:http` — the core of what Express does — so you understand frameworks from the inside.

## Background

### What a router does
A router maps a **method + path** to a handler. Static routes are simple string comparisons. Dynamic routes like `/users/:id/posts/:postId` need pattern matching that extracts **params**. A common technique is to compile each pattern into a regular expression with named groups once, at registration time, and test incoming paths against them in order.

### 404 vs 405
If no route matches the path, respond **404 Not Found**. If the path matches but not with this method, respond **405 Method Not Allowed** and include an `Allow` header listing the valid methods. Many hand-written routers get this wrong.

### Middleware as composition
A middleware is a function that runs before (and optionally after) the handler: logging, authentication, body parsing, CORS. The cleanest async design is **onion-style** composition (as in Koa): each middleware receives `ctx` and `next`, and `await next()` runs everything further in. Code after `await next()` runs on the way back out — perfect for timing and response headers.

```
request → logger → auth → bodyParser → handler
response ← logger ← auth ← bodyParser ←
```

### Centralised error handling
Handlers should `throw` (or reject) and let one place turn errors into responses. Distinguish **expected** errors (an `HttpError` with a status and safe message) from **unexpected** ones (log the stack, respond 500 with a generic message).

### Decoding path segments
Paths arrive percent-encoded (`/files/my%20doc.txt`). Decode params with `decodeURIComponent` — inside a `try`, because malformed encodings (`%E0%A4%A`) throw `URIError`, which should become a 400.

```js
// src/match.js
export function compile(pattern) {
  const keys = [];
  const source = pattern
    .split('/')
    .map((seg) => {
      if (seg.startsWith(':')) {
        keys.push(seg.slice(1));
        return `(?<${seg.slice(1)}>[^/]+)`;
      }
      if (seg === '*') return '(?<wildcard>.*)';
      return seg.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');   // escape regex characters
    })
    .join('/');
  return { regex: new RegExp(`^${source}/?$`), keys };
}

export function match(compiled, pathname) {
  const m = compiled.regex.exec(pathname);
  if (!m) return null;
  const params = {};
  for (const [k, v] of Object.entries(m.groups ?? {})) params[k] = decodeURIComponent(v);
  return params;
}
```

### 2. The router

```js
// src/router.js
import { compile, match } from './match.js';
import { HttpError } from './errors.js';

export class Router {
  #routes = [];

  add(method, pattern, handler) {
    this.#routes.push({ method, pattern, compiled: compile(pattern), handler });
    return this;
  }
  get(p, h) { return this.add('GET', p, h); }
  post(p, h) { return this.add('POST', p, h); }
  put(p, h) { return this.add('PUT', p, h); }
  patch(p, h) { return this.add('PATCH', p, h); }
  delete(p, h) { return this.add('DELETE', p, h); }

  resolve(method, pathname) {
    const allowed = new Set();
    for (const route of this.#routes) {
      let params;
      try {
        params = match(route.compiled, pathname);
      } catch {
        throw new HttpError(400, 'malformed URL encoding');
      }
      if (!params) continue;
      if (route.method === method || (method === 'HEAD' && route.method === 'GET')) {
        return { handler: route.handler, params };
      }
      allowed.add(route.method);
    }
    if (allowed.size) throw new HttpError(405, 'method not allowed', { allow: [...allowed].join(', ') });
    throw new HttpError(404, 'not found');
  }
}
```

### 3. Onion middleware composition

```js
// src/compose.js
export function compose(middlewares) {
  return function run(ctx) {
    let index = -1;
    const dispatch = (i) => {
      if (i <= index) return Promise.reject(new Error('next() called multiple times'));
      index = i;
      const fn = middlewares[i];
      if (!fn) return Promise.resolve();
      try {
        return Promise.resolve(fn(ctx, () => dispatch(i + 1)));
      } catch (err) {
        return Promise.reject(err);
      }
    };
    return dispatch(0);
  };
}
```

### 4. Middlewares

```js
// src/middleware.js
import { randomUUID } from 'node:crypto';
import { HttpError } from './errors.js';

export async function requestId(ctx, next) {
  ctx.id = ctx.req.headers['x-request-id'] ?? randomUUID();
  ctx.res.setHeader('x-request-id', ctx.id);
  await next();
}

export async function logger(ctx, next) {
  const start = performance.now();
  try {
    await next();
  } finally {
    const ms = (performance.now() - start).toFixed(1);
    console.log(`${ctx.id} ${ctx.req.method} ${ctx.url.pathname} ${ctx.res.statusCode} ${ms}ms`);
  }
}

export function jsonBody({ limit = 100_000 } = {}) {
  return async (ctx, next) => {
    if (['POST', 'PUT', 'PATCH'].includes(ctx.req.method)) {
      if (!ctx.req.headers['content-type']?.startsWith('application/json')) {
        throw new HttpError(415, 'expected application/json');
      }
      const chunks = [];
      let size = 0;
      for await (const chunk of ctx.req) {
        if ((size += chunk.length) > limit) throw new HttpError(413, 'body too large');
        chunks.push(chunk);
      }
      try {
        ctx.body = chunks.length ? JSON.parse(Buffer.concat(chunks)) : {};
      } catch {
        throw new HttpError(400, 'invalid JSON');
      }
    }
    await next();
  };
}

export function requireToken(token) {
  return async (ctx, next) => {
    if (ctx.req.headers.authorization !== `Bearer ${token}`) throw new HttpError(401, 'unauthorized');
    await next();
  };
}
```

### 5. Put it together

```js
// server.js
import { createServer } from 'node:http';
import { Router } from './src/router.js';
import { compose } from './src/compose.js';
import { HttpError } from './src/errors.js';
import { requestId, logger, jsonBody } from './src/middleware.js';

const users = new Map([['1', { id: '1', name: 'Ada' }]]);
let nextId = 2;

const router = new Router()
  .get('/users', (ctx) => ctx.json([...users.values()]))
  .get('/users/:id', (ctx) => {
    const user = users.get(ctx.params.id);
    if (!user) throw new HttpError(404, `user ${ctx.params.id} not found`);
    ctx.json(user);
  })
  .post('/users', (ctx) => {
    if (typeof ctx.body.name !== 'string' || !ctx.body.name.trim()) throw new HttpError(422, 'name is required');
    const user = { id: String(nextId++), name: ctx.body.name.trim() };
    users.set(user.id, user);
    ctx.json(user, 201);
  })
  .delete('/users/:id', (ctx) => {
    if (!users.delete(ctx.params.id)) throw new HttpError(404, 'not found');
    ctx.status(204);
  });

const routeHandler = async (ctx) => {
  const { handler, params } = router.resolve(ctx.req.method, ctx.url.pathname);
  ctx.params = params;
  await handler(ctx);
};

const app = compose([requestId, logger, jsonBody(), routeHandler]);

createServer(async (req, res) => {
  const ctx = {
    req, res,
    url: new URL(req.url, 'http://localhost'),
    json(data, status = 200) {
      res.writeHead(status, { 'content-type': 'application/json' });
      res.end(JSON.stringify(data));
    },
    status(code) { res.writeHead(code); res.end(); },
  };
  try {
    await app(ctx);
    if (!res.writableEnded) throw new HttpError(500, 'handler did not send a response');
  } catch (err) {
    const expected = err instanceof HttpError;
    if (!expected) console.error(err);
    if (res.headersSent) return res.destroy();      // too late to send an error body
    res.writeHead(expected ? err.status : 500, { 'content-type': 'application/json', ...(err.headers ?? {}) });
    res.end(JSON.stringify({ error: expected ? err.message : 'internal server error', requestId: ctx.id }));
  }
}).listen(3000, () => console.log('http://localhost:3000'));
```
