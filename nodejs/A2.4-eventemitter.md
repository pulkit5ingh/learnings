**Type:** Assignment

## Goal
Use and build event-driven APIs with `EventEmitter` — the pattern behind streams, servers, sockets, and `process` itself.

## Background

### The observer pattern, built in
`EventEmitter` from `node:events` lets an object **emit** named events and lets any number of **listeners** react to them. Much of Node is built on it: `http.Server` emits `request`, streams emit `data`/`end`/`error`, `process` emits `exit` and signals.

```js
emitter.on('event', listener);    // add a listener (alias: addListener)
emitter.once('event', listener);  // listener removed after first call
emitter.off('event', listener);   // remove (alias: removeListener) — needs the same function reference
emitter.emit('event', ...args);   // call all listeners, returns true if there were any
```

### Emission is synchronous
`emit()` calls every listener **synchronously, in registration order**, before returning. An event-driven API is not automatically asynchronous. If a listener throws, the exception propagates out of `emit()` to the caller.

### The special `error` event
If an emitter emits `'error'` and **no** `'error'` listener is registered, Node **throws** the error — usually crashing the process. Any emitter that can fail (streams, sockets, your own classes) must have an `error` listener.

### Memory leak warnings
By default, adding more than 10 listeners for the same event prints `MaxListenersExceededWarning`. It almost always means listeners are being added repeatedly (e.g. inside a request handler) and never removed. Raise the limit with `setMaxListeners(n)` only when many listeners are genuinely intended.

### Promise and async-iterator helpers
- `await events.once(emitter, 'name')` — resolves with the event's arguments array; rejects if `'error'` is emitted first. Accepts `{ signal }`.
- `for await (const [arg] of events.on(emitter, 'name'))` — consume events as an async iterator.

### Extending EventEmitter
The idiomatic way to build an event-driven class is `class Thing extends EventEmitter`. Document which events you emit and their arguments; emit `'error'` for failures.

```js
// basics.js
import { EventEmitter } from 'node:events';

const bus = new EventEmitter();

function onOrder(order) { console.log('listener 1: order', order.id); }
bus.on('order', onOrder);
bus.on('order', (order) => console.log('listener 2: total', order.total));
bus.once('order', () => console.log('once: first order only'));

console.log('before emit');
bus.emit('order', { id: 1, total: 30 });
console.log('after emit — all listeners already ran (synchronous)');
bus.emit('order', { id: 2, total: 99 });

bus.off('order', onOrder);
bus.emit('order', { id: 3, total: 5 });   // listener 1 is gone

console.log('listener count:', bus.listenerCount('order'));
console.log('emit with no listeners returns', bus.emit('nobody'));
```

### 2. The error event

```js
// error-event.js
import { EventEmitter } from 'node:events';

const e = new EventEmitter();

e.on('error', (err) => console.log('handled:', err.message));
e.emit('error', new Error('disk full'));

const unguarded = new EventEmitter();
unguarded.emit('error', new Error('nobody is listening'));   // throws → crash
console.log('never printed');
```

### 3. Build an event-driven class

```js
// downloader.js
import { EventEmitter } from 'node:events';
import { setTimeout as sleep } from 'node:timers/promises';

/**
 * Events:
 *  'start'    (url)
 *  'progress' (percent)
 *  'done'     (url, bytes)
 *  'error'    (err)
 */
export class FakeDownloader extends EventEmitter {
  async download(url, { failAt } = {}) {
    this.emit('start', url);
    try {
      for (let pct = 0; pct <= 100; pct += 25) {
        await sleep(100);
        if (failAt !== undefined && pct >= failAt) throw new Error(`connection reset at ${pct}%`);
        this.emit('progress', pct);
      }
      this.emit('done', url, 1024);
    } catch (err) {
      this.emit('error', err);
    }
  }
}
```

```js
// use-downloader.js
import { FakeDownloader } from './downloader.js';

const d = new FakeDownloader();
d.on('start', (url) => console.log('start', url));
d.on('progress', (p) => console.log(`  ${p}%`));
d.on('done', (url, bytes) => console.log('done', url, bytes, 'bytes'));
d.on('error', (err) => console.error('failed:', err.message));

await d.download('https://example.com/a.zip');
await d.download('https://example.com/b.zip', { failAt: 50 });
```

### 4. Await a single event

```js
// once.js
import { once } from 'node:events';
import { createServer } from 'node:http';
import { FakeDownloader } from './downloader.js';

const server = createServer((req, res) => res.end('hi'));
server.listen(0);
await once(server, 'listening');                 // no callback needed
console.log('listening on port', server.address().port);
server.close();

const d = new FakeDownloader();
const done = once(d, 'done');                    // rejects if 'error' fires first
d.download('x', { failAt: 25 });
try {
  await done;
} catch (err) {
  console.log('once rejected with:', err.message);
}

// With a timeout
try {
  await once(new FakeDownloader(), 'done', { signal: AbortSignal.timeout(100) });
} catch (err) {
  console.log('timed out:', err.name);
}
```

### 5. Events as an async iterator

```js
// iterate.js
import { on } from 'node:events';
import { FakeDownloader } from './downloader.js';

const d = new FakeDownloader();
const ac = new AbortController();
d.once('done', () => ac.abort());

d.download('stream.zip');

try {
  for await (const [pct] of on(d, 'progress', { signal: ac.signal })) {
    console.log('progress via iterator:', pct);
  }
} catch (err) {
  if (err.name !== 'AbortError') throw err;
  console.log('iteration ended');
}
```

### 6. Create and fix a listener leak

```js
// leak.js
import { EventEmitter } from 'node:events';

const config = new EventEmitter();

function handleRequest(id) {
  // BUG: a new listener on a long-lived emitter for every request
  config.on('reload', () => console.log(`request ${id} saw reload`));
}

for (let i = 0; i < 15; i++) handleRequest(i);   // MaxListenersExceededWarning
console.log('listeners:', config.listenerCount('reload'));
```

Fix it two ways: (a) use `once` and remove the listener when the request finishes with `config.off`, and (b) restructure so a single listener updates shared state that requests read.
