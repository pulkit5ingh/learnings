# P1.2 — Project Scaffolder CLI

**Phase:** 1 — Node.js Runtime Fundamentals
**Topic:** Interactive prompts, recursive template copying, variable substitution, `npx`-style tools
**Type:** Project

---

## Goal

Build `create-node-app` — a scaffolding tool like `npm create vite`. It asks the user a few questions (or reads flags), copies a template directory, substitutes variables into file contents and names, and prints next steps.

## What to Build

- `create-node-app [dir] [--template api|cli|lib] [--name NAME] [--yes]`
- Interactive prompts for any missing values (skipped with `--yes`, using defaults)
- Three templates bundled with the tool in `templates/`
- `{{name}}`, `{{description}}`, `{{year}}` placeholders replaced in file contents
- Files named `_gitignore` renamed to `.gitignore` (npm strips dotfiles named `.gitignore` when publishing)
- Refuse to scaffold into a non-empty directory unless `--force`
- Optional `--install` runs `npm install` in the new project
- Clear "next steps" output

### Prompts

```js
// src/prompts.js
import { createInterface } from 'node:readline/promises';
import { stdin, stdout } from 'node:process';

export async function ask(questions) {
  const rl = createInterface({ input: stdin, output: stdout });
  const answers = {};
  try {
    for (const q of questions) {
      const suffix = q.choices ? ` (${q.choices.join('/')})` : '';
      while (true) {
        const raw = (await rl.question(`${q.message}${suffix} [${q.default}]: `)).trim();
        const value = raw || q.default;
        if (!q.choices || q.choices.includes(value)) {
          answers[q.name] = value;
          break;
        }
        console.error(`  please choose one of: ${q.choices.join(', ')}`);
      }
    }
  } finally {
    rl.close();
  }
  return answers;
}
```

### Scaffolding

```js
// src/scaffold.js
import { readdir, mkdir, readFile, writeFile, copyFile } from 'node:fs/promises';
import { join } from 'node:path';

const TEXT_EXTENSIONS = new Set(['.js', '.json', '.md', '.txt', '.ts', '', '.yml', '.yaml']);
const RENAMES = { _gitignore: '.gitignore' };

function substitute(text, vars) {
  return text.replace(/\{\{(\w+)\}\}/g, (match, key) => (key in vars ? vars[key] : match));
}

function extname(file) {
  const i = file.lastIndexOf('.');
  return i > 0 ? file.slice(i) : '';
}

export async function scaffold(templateDir, targetDir, vars) {
  const created = [];

  async function walk(src, dest) {
    await mkdir(dest, { recursive: true });
    for (const entry of await readdir(src, { withFileTypes: true })) {
      const outName = RENAMES[entry.name] ?? substitute(entry.name, vars);
      const from = join(src, entry.name);
      const to = join(dest, outName);

      if (entry.isDirectory()) {
        await walk(from, to);
      } else if (TEXT_EXTENSIONS.has(extname(entry.name))) {
        await writeFile(to, substitute(await readFile(from, 'utf8'), vars));
        created.push(to);
      } else {
        await copyFile(from, to); // binary files are copied byte for byte
        created.push(to);
      }
    }
  }

  await walk(templateDir, targetDir);
  return created;
}
```

### Running npm install

```js
// src/install.js
import { spawn } from 'node:child_process';

export function npmInstall(cwd) {
  return new Promise((resolve, reject) => {
    const npm = process.platform === 'win32' ? 'npm.cmd' : 'npm';
    const child = spawn(npm, ['install'], { cwd, stdio: 'inherit', shell: process.platform === 'win32' });
    child.on('error', reject);
    child.on('exit', code => (code === 0 ? resolve() : reject(new Error(`npm install exited with ${code}`))));
  });
}
```

### Entry point

```js
#!/usr/bin/env node
// bin/create.js
import { parseArgs } from 'node:util';
import { readdir } from 'node:fs/promises';
import { resolve, basename, join, relative } from 'node:path';
import { ask } from '../src/prompts.js';
import { scaffold } from '../src/scaffold.js';
import { npmInstall } from '../src/install.js';

const TEMPLATES = ['api', 'cli', 'lib'];
const templatesRoot = join(import.meta.dirname, '..', 'templates');

const { values, positionals } = parseArgs({
  options: {
    template: { type: 'string', short: 't' },
    name:     { type: 'string' },
    yes:      { type: 'boolean', short: 'y' },
    force:    { type: 'boolean' },
    install:  { type: 'boolean' },
  },
  allowPositionals: true,
});

async function isEmptyDir(dir) {
  try {
    return (await readdir(dir)).length === 0;
  } catch (err) {
    if (err.code === 'ENOENT') return true;
    throw err;
  }
}

async function main() {
  const targetDir = resolve(positionals[0] ?? 'my-app');
  const defaults = {
    name: values.name ?? basename(targetDir),
    template: values.template ?? 'api',
    description: 'A Node.js project',
  };

  const answers = values.yes
    ? defaults
    : await ask([
        { name: 'name', message: 'Project name', default: defaults.name },
        { name: 'template', message: 'Template', default: defaults.template, choices: TEMPLATES },
        { name: 'description', message: 'Description', default: defaults.description },
      ]);

  if (!TEMPLATES.includes(answers.template)) {
    console.error(`unknown template "${answers.template}"`);
    return 2;
  }
  if (!values.force && !(await isEmptyDir(targetDir))) {
    console.error(`${targetDir} is not empty (use --force to scaffold anyway)`);
    return 1;
  }

  const vars = { ...answers, year: String(new Date().getFullYear()) };
  const files = await scaffold(join(templatesRoot, answers.template), targetDir, vars);
  files.forEach(f => console.log(`  created ${relative(process.cwd(), f)}`));

  if (values.install) await npmInstall(targetDir);

  console.log(`\nDone. Next steps:\n  cd ${relative(process.cwd(), targetDir) || '.'}`);
  if (!values.install) console.log('  npm install');
  console.log('  npm run dev');
  return 0;
}

process.exitCode = await main();
```
