# P6.1 — Parallel File Hasher

**Phase:** 6 — Performance & Concurrency
**Topic:** Worker pools, batching, streaming hashes inside workers, progress reporting, benchmarking, choosing the bottleneck
**Type:** Project

---

## Goal

Build `phash` — a tool that computes checksums for every file in a directory tree using a pool of worker threads, finds duplicate files, and writes a manifest. Benchmark it against a single-threaded version and explain the results: when does parallelism help, and when is the disk the bottleneck?

## What to Build

- `phash <dir> [--algo sha256|sha1|md5] [--workers N] [--out manifest.json] [--dupes]`
- Walk the tree on the main thread; hash files in workers (each worker streams its file through `createHash`)
- Work is dispatched in **batches** of small files to reduce messaging overhead; large files go individually
- Live progress on stderr: files/sec, MB/sec, ETA
- Output: manifest of `{ path, size, hash }`, sorted by path
- `--dupes`: group files with identical hashes and report wasted bytes
- `--workers 0` runs single-threaded on the main thread (for benchmarking)
- `--verify manifest.json` re-hashes and reports changed, missing, and new files
- A benchmark script comparing 0, 1, 2, 4, and N workers on CPU-heavy (sha256 over cached data) vs I/O-heavy scenarios
- Ctrl+C terminates workers and exits cleanly

## Implementation

### Shared hashing function

```js
// src/hashFile.js
import { createReadStream } from 'node:fs';
import { createHash } from 'node:crypto';
import { pipeline } from 'node:stream/promises';

export async function hashFile(path, algo = 'sha256') {
  const h = createHash(algo);
  await pipeline(createReadStream(path, { highWaterMark: 1024 * 1024 }), h);
  return h.digest('hex');
}
```

### Worker

```js
// src/hash-worker.js
import { parentPort } from 'node:worker_threads';
import { hashFile } from './hashFile.js';

parentPort.on('message', async ({ batch, algo }) => {
  const results = [];
  for (const file of batch) {
    try {
      results.push({ ...file, hash: await hashFile(file.path, algo) });
    } catch (err) {
      results.push({ ...file, error: err.code ?? err.message });   // unreadable file — report, don't crash
    }
  }
  parentPort.postMessage(results);
});
```

Note that this worker is async: while one file's stream is waiting for the disk, the worker's own event loop is idle. You could hash several files concurrently inside each worker — try it as an extension.

### Batching

```js
// src/batcher.js
export async function* batches(files, { maxFiles = 64, maxBytes = 32 * 1024 * 1024 } = {}) {
  let batch = [], bytes = 0;
  for await (const f of files) {
    if (f.size >= maxBytes) { yield [f]; continue; }            // big file: its own batch
    batch.push(f);
    bytes += f.size;
    if (batch.length >= maxFiles || bytes >= maxBytes) { yield batch; batch = []; bytes = 0; }
  }
  if (batch.length) yield batch;
}
```

Sending 100,000 single-file messages would spend more time on messaging than on hashing small files.

### Walking

```js
// src/walk.js
import { readdir, stat } from 'node:fs/promises';
import { join } from 'node:path';

export async function* walk(dir, ignore = new Set(['.git', 'node_modules'])) {
  for (const e of await readdir(dir, { withFileTypes: true })) {
    if (ignore.has(e.name)) continue;
    const p = join(dir, e.name);
    if (e.isDirectory()) yield* walk(p, ignore);
    else if (e.isFile()) yield { path: p, size: (await stat(p)).size };
  }
}
```

### Progress

```js
// src/progress.js
export class Progress {
  files = 0; bytes = 0; totalFiles = 0; totalBytes = 0; start = performance.now();

  render() {
    const secs = (performance.now() - this.start) / 1000;
    const mbps = this.bytes / 1e6 / secs;
    const eta = this.totalBytes && mbps ? Math.max(0, (this.totalBytes - this.bytes) / 1e6 / mbps) : 0;
    process.stderr.write(`\r${this.files}/${this.totalFiles || '?'} files  ${mbps.toFixed(1)} MB/s  ${(this.files / secs).toFixed(0)} files/s  ETA ${eta.toFixed(0)}s   `);
  }
}
```

### Orchestration

```js
#!/usr/bin/env node
// bin/phash.js
import { parseArgs } from 'node:util';
import { availableParallelism } from 'node:os';
import { writeFile, readFile } from 'node:fs/promises';
import { walk } from '../src/walk.js';
import { batches } from '../src/batcher.js';
import { WorkerPool } from '../src/pool.js';
import { hashFile } from '../src/hashFile.js';
import { Progress } from '../src/progress.js';
import { findDupes, diffManifests } from '../src/report.js';

const { values, positionals } = parseArgs({
  options: {
    algo: { type: 'string', default: 'sha256' },
    workers: { type: 'string', default: String(availableParallelism()) },
    out: { type: 'string' },
    dupes: { type: 'boolean' },
    verify: { type: 'string' },
  },
  allowPositionals: true,
});

const dir = positionals[0];
if (!dir) { console.error('usage: phash <dir> [--workers N] [--algo A] [--out F] [--dupes] [--verify F]'); process.exit(2); }
const workerCount = Number(values.workers);

// Pass 1: collect file list (cheap metadata) so we can show totals and ETA
const files = [];
for await (const f of walk(dir)) files.push(f);
const progress = new Progress();
progress.totalFiles = files.length;
progress.totalBytes = files.reduce((n, f) => n + f.size, 0);
const ticker = setInterval(() => progress.render(), 250);

const pool = workerCount > 0 ? new WorkerPool(new URL('../src/hash-worker.js', import.meta.url), workerCount) : null;
process.once('SIGINT', async () => { clearInterval(ticker); await pool?.close(); process.stderr.write('\ncancelled\n'); process.exit(130); });

const results = [];
const record = (r) => { results.push(r); progress.files++; progress.bytes += r.size; };

if (pool) {
  const inflight = [];
  for await (const batch of batches(files)) {
    inflight.push(pool.run({ batch, algo: values.algo }).then((rs) => rs.forEach(record)));
  }
  await Promise.all(inflight);
  await pool.close();
} else {
  for (const f of files) {
    try { record({ ...f, hash: await hashFile(f.path, values.algo) }); }
    catch (err) { record({ ...f, error: err.code }); }
  }
}

clearInterval(ticker);
progress.render();
process.stderr.write(`\nhashed ${results.length} files (${(progress.totalBytes / 1e6).toFixed(1)} MB) in ${((performance.now() - progress.start) / 1000).toFixed(2)}s with ${workerCount} workers\n`);

results.sort((a, b) => a.path.localeCompare(b.path));
const manifest = { algo: values.algo, root: dir, createdAt: new Date().toISOString(), files: results };

if (values.out) await writeFile(values.out, JSON.stringify(manifest, null, 2));
if (values.dupes) {
  for (const group of findDupes(results)) console.log(`${group.hash.slice(0, 12)}  ${group.files.length}× ${group.size} B  wasted ${group.wasted} B\n    ${group.files.join('\n    ')}`);
}
if (values.verify) {
  const diff = diffManifests(JSON.parse(await readFile(values.verify, 'utf8')), manifest);
  console.log(JSON.stringify(diff, null, 2));
  if (diff.changed.length || diff.missing.length) process.exitCode = 1;
}
```

The pool's `run()` posts `{ batch, algo }` to an idle worker and resolves with the worker's reply — reuse your `WorkerPool` from A6.2 with a simpler message handler (`worker.on('message', resolve)`).

### Reports

```js
// src/report.js
export function findDupes(files) {
  const byHash = new Map();
  for (const f of files) if (f.hash && f.size > 0) (byHash.get(f.hash) ?? byHash.set(f.hash, []).get(f.hash)).push(f);
  return [...byHash.values()]
    .filter((g) => g.length > 1)
    .map((g) => ({ hash: g[0].hash, size: g[0].size, files: g.map((f) => f.path), wasted: g[0].size * (g.length - 1) }))
    .sort((a, b) => b.wasted - a.wasted);
}

export function diffManifests(before, after) {
  const old = new Map(before.files.map((f) => [f.path, f.hash]));
  const now = new Map(after.files.map((f) => [f.path, f.hash]));
  return {
    changed: [...now].filter(([p, h]) => old.has(p) && old.get(p) !== h).map(([p]) => p),
    missing: [...old.keys()].filter((p) => !now.has(p)),
    added: [...now.keys()].filter((p) => !old.has(p)),
  };
}
```

### Benchmark

```js
// scripts/bench.js
import { execFileSync } from 'node:child_process';
import { availableParallelism } from 'node:os';

const dir = process.argv[2];
const counts = [...new Set([0, 1, 2, 4, availableParallelism()])];
execFileSync(process.execPath, ['bin/phash.js', dir, '--workers', '0'], { stdio: 'ignore' });   // warm the OS file cache

for (const algo of ['sha256', 'md5']) {
  for (const n of counts) {
    const t = performance.now();
    execFileSync(process.execPath, ['bin/phash.js', dir, '--workers', String(n), '--algo', algo], { stdio: 'ignore' });
    console.log(`${algo.padEnd(6)} workers=${String(n).padStart(2)}  ${Math.round(performance.now() - t)}ms`);
  }
}
```
