## Goal
Build HTTP servers with nothing but `node:http`, and understand every object and event that frameworks like Express wrap.

## Background

### The server and its events
`http.createServer(handler)` returns an `http.Server` (an EventEmitter). The handler is attached to its `'request'` event and is called with two streams:

- **`req`** — an `http.IncomingMessage`, a **Readable** stream. Properties: `req.method`, `req.url` (path + query string only, e.g. `/users?page=2`), `req.headers` (lower-cased keys), `req.httpVersion`, `req.socket`.
- **`res`** — an `http.ServerResponse`, a **Writable** stream. You set a status and headers, write zero or more body chunks, and must call `res.end()` exactly once.

Other server events: `'listening'`, `'connection'` (a new TCP socket), `'close'`, `'error'` (e.g. `EADDRINUSE`), `'clientError'` (malformed requests).

### Writing a response
```js
res.statusCode = 200;                       // or res.writeHead(200, headers)
res.setHeader('content-type', 'application/json');
res.write('partial body…');                 // optional, can be called many times
res.end('final chunk');                     // required
```
Headers must be set **before** the first `write`/`end` — afterwards `res.headersSent` is `true` and `setHeader` throws `ERR_HTTP_HEADERS_SENT`. If you never call `res.end()`, the client hangs until a timeout.

### Parsing the URL
`req.url` is a relative string. Parse it with the WHATWG `URL` class: `new URL(req.url, 'http://localhost')`, then use `url.pathname` and `url.searchParams`.

### The body is a stream
Node does not buffer request bodies for you. You read `req` as a stream (`for await (const chunk of req)`), concatenate the Buffers, and parse. You must also **limit body size** — otherwise one client can send gigabytes and exhaust your memory.

### Connections, keep-alive, and timeouts
HTTP/1.1 keeps TCP connections open between requests by default. The server has several timeouts:
- `server.requestTimeout` — max time to receive the whole request (default 300 s)
- `server.headersTimeout` — max time to receive headers (default 60 s)
- `server.keepAliveTimeout` — how long an idle keep-alive connection stays open (default 5 s)

`server.close()` stops accepting new connections but waits for existing ones; `server.closeIdleConnections()` and `server.closeAllConnections()` help during shutdown.

### Status codes that matter
200 OK · 201 Created · 204 No Content · 301/302/307/308 redirects · 304 Not Modified · 400 Bad Request · 401 Unauthorized · 403 Forbidden · 404 Not Found · 405 Method Not Allowed · 409 Conflict · 413 Payload Too Large · 415 Unsupported Media Type · 422 Unprocessable Entity · 429 Too Many Requests · 500 Internal Server Error · 503 Service Unavailable.

### 2. Inspect the request

```js
// inspect.js
import { createServer } from 'node:http';

createServer((req, res) => {
  const url = new URL(req.url, `http://${req.headers.host}`);
  const info = {
    method: req.method,
    rawUrl: req.url,
    pathname: url.pathname,
    query: Object.fromEntries(url.searchParams),
    headers: req.headers,
    httpVersion: req.httpVersion,
    remote: `${req.socket.remoteAddress}:${req.socket.remotePort}`,
  };
  res.writeHead(200, { 'content-type': 'application/json' });
  res.end(JSON.stringify(info, null, 2));
}).listen(3000);
```

### 3. Read a JSON body safely

```js
// body.js
import { createServer } from 'node:http';

const MAX_BYTES = 1024 * 1024;   // 1 MB

class HttpError extends Error {
  constructor(status, message) { super(message); this.status = status; }
}

async function readJson(req) {
  if (!req.headers['content-type']?.startsWith('application/json')) {
    throw new HttpError(415, 'content-type must be application/json');
  }
  const declared = Number(req.headers['content-length'] ?? 0);
  if (declared > MAX_BYTES) throw new HttpError(413, 'body too large');

  const chunks = [];
  let size = 0;
  for await (const chunk of req) {
    size += chunk.length;
    if (size > MAX_BYTES) throw new HttpError(413, 'body too large');   // content-length can lie
    chunks.push(chunk);
  }
  try {
    return JSON.parse(Buffer.concat(chunks).toString('utf8'));
  } catch {
    throw new HttpError(400, 'invalid JSON');
  }
}

function send(res, status, data) {
  res.writeHead(status, { 'content-type': 'application/json' });
  res.end(JSON.stringify(data));
}

createServer(async (req, res) => {
  try {
    if (req.method !== 'POST') {
      res.setHeader('allow', 'POST');
      return send(res, 405, { error: 'method not allowed' });
    }
    const body = await readJson(req);
    send(res, 201, { received: body });
  } catch (err) {
    send(res, err.status ?? 500, { error: err.status ? err.message : 'internal error' });
    if (!err.status) console.error(err);
  }
}).listen(3000);
```

```bash
curl -X POST localhost:3000 -H 'content-type: application/json' -d '{"name":"ada"}'
curl -X POST localhost:3000 -H 'content-type: application/json' -d '{bad'
curl -X POST localhost:3000 -d 'x=1'

```

### 4. Headers-sent errors and streaming responses

```js
// stream-response.js
import { createServer } from 'node:http';
import { setTimeout as sleep } from 'node:timers/promises';

createServer(async (req, res) => {
  if (req.url === '/oops') {
    res.end('done');
    try { res.setHeader('x-late', '1'); } catch (err) { console.log(err.code); }   // ERR_HTTP_HEADERS_SENT
    return;
  }
  res.writeHead(200, { 'content-type': 'text/plain' });   // no content-length → chunked encoding
  for (let i = 1; i <= 5; i++) {
    res.write(`chunk ${i}\n`);
    await sleep(500);
  }
  res.end('end\n');
}).listen(3000);
```

```bash
curl -N localhost:3000        # chunks arrive one by one
curl -v localhost:3000 2>&1 | grep -i transfer-encoding
curl localhost:3000/oops
```

### 5. Handle client disconnects

```js
// disconnect.js
import { createServer } from 'node:http';

createServer((req, res) => {
  const timer = setInterval(() => res.write(`tick ${Date.now()}\n`), 200);
  res.on('close', () => {
    clearInterval(timer);
    console.log(`connection closed, finished=${res.writableFinished}`);
  });
}).listen(3000);
```

Run `curl -N localhost:3000`, press Ctrl+C, and confirm the interval is cleared.

### 6. Server errors, timeouts, and keep-alive

```js
// server-config.js
import { createServer } from 'node:http';

const server = createServer((req, res) => res.end(`served on socket ${req.socket.remotePort}\n`));

server.keepAliveTimeout = 2000;
server.headersTimeout = 5000;
server.requestTimeout = 10_000;

server.on('error', (err) => {
  if (err.code === 'EADDRINUSE') console.error('port 3000 is already in use');
  else throw err;
  process.exitCode = 1;
});
server.on('connection', (socket) => console.log('new TCP connection from port', socket.remotePort));

server.listen(3000);
```

```bash
curl localhost:3000 localhost:3000          # two requests, one connection (same port)
node server-config.js                        # in a second terminal → EADDRINUSE
```
