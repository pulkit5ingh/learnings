# P4.2 — Static File Server

**Phase:** 4 — Networking & HTTP
**Topic:** Streaming files over HTTP, MIME types, path traversal protection, ETags, range requests, compression, directory listings
**Type:** Project

---

## Goal

Build `serve` — a production-quality static file server like `npx serve` or nginx's static module, using only Node. It must be **safe** (no path traversal), **efficient** (streams files, supports caching and partial content), and **correct** (proper status codes and headers).

## What to Build

- `serve [dir] [--port 8080] [--spa] [--no-listing]`
- Streams files with correct `Content-Type` and `Content-Length`
- Serves `index.html` for directories; HTML directory listing otherwise (unless `--no-listing`)
- Blocks path traversal (`/../../etc/passwd`, encoded variants, null bytes) and dotfiles (`/.env`)
- `ETag` and `Last-Modified`; answers `If-None-Match` / `If-Modified-Since` with `304`
- `Range` requests (`bytes=0-1023`, `bytes=-500`) → `206 Partial Content`, invalid → `416`
- On-the-fly gzip for text types when the client sends `Accept-Encoding: gzip`
- `Cache-Control` long-lived for fingerprinted assets (`app.3f9a2c.js`), `no-cache` otherwise
- `--spa`: unknown paths without an extension fall back to `index.html`
- `HEAD` support; only `GET`/`HEAD` allowed (`405` otherwise)
- Access log line per request

### Safe path resolution

```js
// src/resolve.js
import { resolve, sep, normalize } from 'node:path';

export function safeResolve(root, urlPath) {
  let decoded;
  try {
    decoded = decodeURIComponent(urlPath);
  } catch {
    return { error: 400 };
  }
  if (decoded.includes('\0')) return { error: 400 };
  if (decoded.split('/').some(seg => seg.startsWith('.') && seg !== '.')) return { error: 404 };  // dotfiles & '..'

  const full = resolve(root, '.' + normalize('/' + decoded));
  if (full !== root && !full.startsWith(root + sep)) return { error: 403 };
  return { path: full };
}
```

Three layers of defence: decode once and reject null bytes, reject any segment starting with `.` (which covers `..`), and finally verify the resolved absolute path is still inside `root`.

### Conditional requests

```js
// src/conditional.js
export function etagFor(stat) {
  return `W/"${stat.size.toString(16)}-${Math.floor(stat.mtimeMs).toString(16)}"`;
}

export function isNotModified(req, etag, mtime) {
  const inm = req.headers['if-none-match'];
  if (inm) return inm.split(',').map(s => s.trim()).includes(etag) || inm.trim() === '*';
  const ims = req.headers['if-modified-since'];
  if (ims) {
    const since = Date.parse(ims);
    return !Number.isNaN(since) && Math.floor(mtime.getTime() / 1000) <= Math.floor(since / 1000);
  }
  return false;
}
```

### Range parsing

```js
// src/range.js
// Returns null (no/ignored range), { invalid: true }, or { start, end }
export function parseRange(header, size) {
  if (!header) return null;
  const m = /^bytes=(\d*)-(\d*)$/.exec(header.trim());
  if (!m) return null;                         // multi-range or unknown unit → serve the full file
  let [, s, e] = m;
  let start, end;
  if (s === '' && e === '') return { invalid: true };
  if (s === '') {                              // suffix: last N bytes
    start = Math.max(size - Number(e), 0);
    end = size - 1;
  } else {
    start = Number(s);
    end = e === '' ? size - 1 : Math.min(Number(e), size - 1);
  }
  if (start > end || start >= size) return { invalid: true };
  return { start, end };
}
```

### Directory listing

```js
// src/listing.js
import { readdir } from 'node:fs/promises';

const escape = (s) => s.replace(/[&<>"']/g, c => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' })[c]);

export async function renderListing(dir, urlPath) {
  const entries = (await readdir(dir, { withFileTypes: true }))
    .filter(e => !e.name.startsWith('.'))
    .sort((a, b) => (b.isDirectory() - a.isDirectory()) || a.name.localeCompare(b.name));
  const base = urlPath.endsWith('/') ? urlPath : `${urlPath}/`;
  const items = entries.map(e => {
    const name = e.isDirectory() ? `${e.name}/` : e.name;
    return `<li><a href="${escape(base + encodeURIComponent(e.name))}${e.isDirectory() ? '/' : ''}">${escape(name)}</a></li>`;
  });
  return `<!doctype html><meta charset="utf-8"><title>Index of ${escape(base)}</title>
<h1>Index of ${escape(base)}</h1><ul>${base !== '/' ? '<li><a href="../">../</a></li>' : ''}${items.join('')}</ul>`;
}
```

File names are HTML-escaped so a file named `<script>.txt` can't inject markup.

### The request flow

```js
// src/server.js
import { createServer } from 'node:http';
import { createReadStream } from 'node:fs';
import { stat } from 'node:fs/promises';
import { extname, join } from 'node:path';
import { createGzip } from 'node:zlib';
import { pipeline } from 'node:stream/promises';
import { safeResolve } from './resolve.js';
import { mimeFor, isCompressible } from './mime.js';
import { etagFor, isNotModified } from './conditional.js';
import { parseRange } from './range.js';
import { renderListing } from './listing.js';

const FINGERPRINT = /\.[0-9a-f]{6,}\.\w+$/;

async function statOrNull(p) {
  try { return await stat(p); } catch (err) { if (['ENOENT', 'ENOTDIR'].includes(err.code)) return null; throw err; }
}

export function createStaticServer(root, { spa = false, listing = true } = {}) {
  const send = (res, status, body = '', headers = {}) => {
    res.writeHead(status, { 'content-type': 'text/plain; charset=utf-8', ...headers });
    res.end(body);
  };

  async function handle(req, res) {
    if (!['GET', 'HEAD'].includes(req.method)) return send(res, 405, 'Method Not Allowed\n', { allow: 'GET, HEAD' });

    const urlPath = new URL(req.url, 'http://x').pathname;
    const resolved = safeResolve(root, urlPath);
    if (resolved.error) return send(res, resolved.error, `${resolved.error}\n`);

    let filePath = resolved.path;
    let s = await statOrNull(filePath);

    if (s?.isDirectory()) {
      if (!urlPath.endsWith('/')) return send(res, 301, '', { location: `${urlPath}/` });
      const index = join(filePath, 'index.html');
      const indexStat = await statOrNull(index);
      if (indexStat) { filePath = index; s = indexStat; }
      else if (listing) return send(res, 200, req.method === 'HEAD' ? '' : await renderListing(filePath, urlPath), { 'content-type': 'text/html; charset=utf-8' });
      else return send(res, 404, 'Not Found\n');
    }

    if (!s && spa && !extname(urlPath)) {
      filePath = join(root, 'index.html');
      s = await statOrNull(filePath);
    }
    if (!s?.isFile()) return send(res, 404, 'Not Found\n');

    const type = mimeFor(extname(filePath));
    const etag = etagFor(s);
    const headers = {
      'content-type': type,
      etag,
      'last-modified': s.mtime.toUTCString(),
      'accept-ranges': 'bytes',
      'cache-control': FINGERPRINT.test(filePath) ? 'public, max-age=31536000, immutable' : 'no-cache',
      vary: 'Accept-Encoding',
    };

    if (isNotModified(req, etag, s.mtime)) { res.writeHead(304, headers); return res.end(); }

    const range = parseRange(req.headers.range, s.size);
    if (range?.invalid) return send(res, 416, '', { 'content-range': `bytes */${s.size}` });

    if (range) {
      res.writeHead(206, { ...headers, 'content-range': `bytes ${range.start}-${range.end}/${s.size}`, 'content-length': range.end - range.start + 1 });
      if (req.method === 'HEAD') return res.end();
      return pipeline(createReadStream(filePath, { start: range.start, end: range.end }), res);
    }

    const gzip = isCompressible(type) && s.size > 1024 && /\bgzip\b/.test(req.headers['accept-encoding'] ?? '');
    if (gzip) res.writeHead(200, { ...headers, 'content-encoding': 'gzip' });     // length unknown → chunked
    else res.writeHead(200, { ...headers, 'content-length': s.size });
    if (req.method === 'HEAD') return res.end();

    const stages = [createReadStream(filePath)];
    if (gzip) stages.push(createGzip());
    return pipeline(...stages, res);
  }

  return createServer(async (req, res) => {
    const start = performance.now();
    res.on('finish', () => console.log(`${req.method} ${req.url} ${res.statusCode} ${(performance.now() - start).toFixed(1)}ms`));
    try {
      await handle(req, res);
    } catch (err) {
      if (err.code === 'ERR_STREAM_PREMATURE_CLOSE') return;       // client disconnected
      console.error(err);
      if (!res.headersSent) send(res, 500, 'Internal Server Error\n');
      else res.destroy();
    }
  });
}
```

### CLI

```js
#!/usr/bin/env node
// bin/serve.js
import { parseArgs } from 'node:util';
import { resolve } from 'node:path';
import { realpath } from 'node:fs/promises';
import { createStaticServer } from '../src/server.js';

const { values, positionals } = parseArgs({
  options: {
    port: { type: 'string', short: 'p', default: '8080' },
    spa: { type: 'boolean', default: false },
    'no-listing': { type: 'boolean', default: false },
  },
  allowPositionals: true,
});

const root = await realpath(resolve(positionals[0] ?? '.'));   // resolve symlinks in the root itself
const server = createStaticServer(root, { spa: values.spa, listing: !values['no-listing'] });
server.listen(Number(values.port), () => console.log(`serving ${root} on http://localhost:${values.port}`));
```

## Deliverables

- [ ] Every traversal attempt (`..`, encoded, null byte) and every dotfile is refused
- [ ] Files stream with the correct MIME type and `Content-Length`
- [ ] Directories redirect to a trailing slash, then serve `index.html` or an escaped listing
- [ ] `ETag` / `If-None-Match` and `If-Modified-Since` produce `304`
- [ ] Range requests return `206` with `Content-Range`; invalid ranges return `416`
- [ ] Text files are gzipped only when the client accepts it; `Vary: Accept-Encoding` is set
- [ ] Fingerprinted assets get `immutable` caching; other files get `no-cache`
- [ ] `HEAD` returns headers without a body; other methods return `405`
- [ ] Client disconnects mid-download don't crash or leak file descriptors
