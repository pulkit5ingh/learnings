# P7.1 — Tested Utility Library

**Phase:** 7 — Testing & Debugging
**Topic:** Test-first design, `node:test`, mock timers, edge cases, coverage thresholds, CI
**Type:** Project

---

## Goal

Build `@you/async-utils` — a small library of async utilities that real services need — and test it thoroughly with `node:test`: 90%+ line and branch coverage, deterministic timing tests with mock timers, and a CI workflow that enforces the threshold.

## What to Build

| Utility | Behaviour |
|---------|-----------|
| `retry(fn, opts)` | Retries with exponential backoff + jitter; `retries`, `minDelay`, `maxDelay`, `shouldRetry(err)`, `signal` |
| `timeout(promise, ms)` | Rejects with `TimeoutError` after `ms`; clears its timer |
| `debounce(fn, ms)` | Trailing debounce with `.cancel()` and `.flush()` |
| `pLimit(n)` | Returns `limit(fn)` running at most `n` tasks concurrently |
| `LRUCache` | `get`/`set`/`has`/`delete`, max size, optional TTL per entry, evicts least recently used |
| `TokenBucket` | Rate limiter: `capacity`, `refillPerSec`, `tryRemove(n)` |

- Every utility has its own test file covering normal behaviour, edge cases, and errors
- No real waiting in tests: all timing via mock timers or injected clocks
- `npm test` fails below 90% line / 90% branch coverage
- GitHub Actions runs tests on Node 22 and 24

### retry

```js
// src/retry.js
const sleepWithSignal = (ms, signal) => new Promise((resolve, reject) => {
  if (signal?.aborted) return reject(signal.reason);
  const t = setTimeout(resolve, ms);
  signal?.addEventListener('abort', () => { clearTimeout(t); reject(signal.reason); }, { once: true });
});

export async function retry(fn, { retries = 3, minDelay = 100, maxDelay = 5000, factor = 2, jitter = true, shouldRetry = () => true, signal, random = Math.random } = {}) {
  for (let attempt = 0; ; attempt++) {
    signal?.throwIfAborted();
    try {
      return await fn(attempt);
    } catch (err) {
      if (attempt >= retries || !shouldRetry(err, attempt)) throw err;
      const base = Math.min(minDelay * factor ** attempt, maxDelay);
      await sleepWithSignal(jitter ? base / 2 + random() * (base / 2) : base, signal);
    }
  }
}
```

### timeout

```js
// src/timeout.js
export class TimeoutError extends Error { name = 'TimeoutError'; }

export function timeout(promise, ms, message = `timed out after ${ms}ms`) {
  let timer;
  return Promise.race([
    promise,
    new Promise((_, reject) => { timer = setTimeout(() => reject(new TimeoutError(message)), ms); }),
  ]).finally(() => clearTimeout(timer));
}
```

### debounce

```js
// src/debounce.js
export function debounce(fn, ms) {
  let timer = null, lastArgs, lastThis;
  function debounced(...args) {
    lastArgs = args; lastThis = this;
    clearTimeout(timer);
    timer = setTimeout(invoke, ms);
  }
  function invoke() { timer = null; fn.apply(lastThis, lastArgs); }
  debounced.cancel = () => { clearTimeout(timer); timer = null; };
  debounced.flush = () => { if (timer) { clearTimeout(timer); invoke(); } };
  debounced.pending = () => timer !== null;
  return debounced;
}
```

### pLimit

```js
// src/pLimit.js
export function pLimit(concurrency) {
  if (!Number.isInteger(concurrency) || concurrency < 1) throw new RangeError('concurrency must be a positive integer');
  let active = 0;
  const queue = [];
  const next = () => {
    if (active >= concurrency || queue.length === 0) return;
    active++;
    const { fn, resolve, reject } = queue.shift();
    Promise.resolve().then(fn).then(resolve, reject).finally(() => { active--; next(); });
  };
  const limit = (fn) => new Promise((resolve, reject) => { queue.push({ fn, resolve, reject }); next(); });
  Object.defineProperties(limit, { active: { get: () => active }, pending: { get: () => queue.length } });
  return limit;
}
```

### LRU cache

```js
// src/lru.js
export class LRUCache {
  #max; #map = new Map(); #now;

  constructor({ max = 100, now = Date.now } = {}) {
    if (!(max >= 1)) throw new RangeError('max must be >= 1');
    this.#max = max; this.#now = now;
  }

  get size() { return this.#map.size; }

  get(key) {
    const entry = this.#map.get(key);
    if (!entry) return undefined;
    if (entry.expires !== undefined && entry.expires <= this.#now()) { this.#map.delete(key); return undefined; }
    this.#map.delete(key);           // move to most-recently-used position
    this.#map.set(key, entry);
    return entry.value;
  }

  has(key) { return this.get(key) !== undefined; }

  set(key, value, { ttlMs } = {}) {
    this.#map.delete(key);
    this.#map.set(key, { value, expires: ttlMs === undefined ? undefined : this.#now() + ttlMs });
    if (this.#map.size > this.#max) this.#map.delete(this.#map.keys().next().value);   // oldest = first
    return this;
  }

  delete(key) { return this.#map.delete(key); }
  clear() { this.#map.clear(); }
}
```

A `Map` preserves insertion order, so deleting and re-inserting on access keeps the least recently used key first.

### Token bucket

```js
// src/tokenBucket.js
export class TokenBucket {
  #tokens; #last;
  constructor({ capacity, refillPerSec, now = Date.now }) {
    this.capacity = capacity; this.refillPerSec = refillPerSec; this.now = now;
    this.#tokens = capacity; this.#last = now();
  }
  #refill() {
    const t = this.now();
    this.#tokens = Math.min(this.capacity, this.#tokens + ((t - this.#last) / 1000) * this.refillPerSec);
    this.#last = t;
  }
  tryRemove(n = 1) {
    this.#refill();
    if (this.#tokens >= n) { this.#tokens -= n; return true; }
    return false;
  }
  get tokens() { this.#refill(); return this.#tokens; }
}
```

### Example tests

```js
// test/retry.test.js
import { describe, it } from 'node:test';
import assert from 'node:assert/strict';
import { retry } from '../src/retry.js';

describe('retry', () => {
  it('returns the first success', async (t) => {
    const fn = t.mock.fn(async () => 'ok');
    assert.equal(await retry(fn), 'ok');
    assert.equal(fn.mock.callCount(), 1);
  });

  it('retries with exponential backoff', async (t) => {
    t.mock.timers.enable({ apis: ['setTimeout'] });
    const fn = t.mock.fn(async (n) => { if (n < 2) throw new Error('flaky'); return n; });
    const p = retry(fn, { minDelay: 100, jitter: false });
    for (const ms of [100, 200]) {
      await new Promise(setImmediate);        // let the rejection be processed
      t.mock.timers.tick(ms);
    }
    assert.equal(await p, 2);
    assert.equal(fn.mock.callCount(), 3);
  });

  it('stops when shouldRetry returns false', async () => {
    let calls = 0;
    await assert.rejects(retry(async () => { calls++; throw new Error('fatal'); }, { shouldRetry: () => false }), /fatal/);
    assert.equal(calls, 1);
  });

  it('throws the last error after exhausting retries', async () => {
    await assert.rejects(retry(async (n) => { throw new Error(`fail ${n}`); }, { retries: 2, minDelay: 0 }), /fail 2/);
  });

  it('aborts while waiting', async () => {
    const ac = new AbortController();
    const p = retry(async () => { throw new Error('x'); }, { minDelay: 10_000, signal: ac.signal });
    setTimeout(() => ac.abort(new Error('cancelled')), 10);
    await assert.rejects(p, /cancelled/);
  });
});
```

```js
// test/lru.test.js
import { it } from 'node:test';
import assert from 'node:assert/strict';
import { LRUCache } from '../src/lru.js';

it('evicts the least recently used entry', () => {
  const c = new LRUCache({ max: 2 });
  c.set('a', 1).set('b', 2);
  c.get('a');                      // a is now most recent
  c.set('c', 3);                   // evicts b
  assert.equal(c.has('b'), false);
  assert.deepEqual([c.get('a'), c.get('c')], [1, 3]);
});

it('expires entries by TTL using an injected clock', () => {
  let now = 0;
  const c = new LRUCache({ max: 10, now: () => now });
  c.set('k', 'v', { ttlMs: 1000 });
  now = 999; assert.equal(c.get('k'), 'v');
  now = 1000; assert.equal(c.get('k'), undefined);
  assert.equal(c.size, 0);
});

it('rejects an invalid max', () => assert.throws(() => new LRUCache({ max: 0 }), RangeError));
```
