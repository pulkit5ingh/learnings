# A6.2 — Worker Threads

**Phase:** 6 — Performance & Concurrency
**Topic:** CPU-bound work, `Worker`, `postMessage`, structured clone, transferables, `SharedArrayBuffer` + `Atomics`, worker pools
**Type:** Assignment

---

## Goal
Move CPU-heavy JavaScript off the main thread so servers stay responsive, and build a reusable worker pool.

## Background

### When you need workers
Node's async I/O already uses other threads, so **I/O-bound** work never needs workers. **CPU-bound** JavaScript — image processing, compression in JS, parsing huge JSON, hashing in loops, PDF generation, ML inference, heavy regex — runs on the main thread and blocks everything (A2.3). Worker threads run JavaScript **in parallel** on other CPU cores.

### What a worker is
`new Worker(file)` starts a separate V8 isolate with its own event loop, heap, and globals, running inside the same process. Workers don't share regular JavaScript objects with the main thread. They communicate by:

- **Messages** — `worker.postMessage(value)` / `parentPort.on('message')`. Values are copied using the **structured clone** algorithm (objects, arrays, Maps, Dates, Buffers, typed arrays… but not functions or class instances' methods).
- **Transfer** — `postMessage(buf, [buf.buffer])` moves an `ArrayBuffer` to the other thread with **zero copying**; the sender can no longer use it.
- **Shared memory** — a `SharedArrayBuffer` is visible to all threads at once; coordinate with `Atomics` (`add`, `load`, `wait`, `notify`) to avoid data races.

`workerData` passes initial data; `parentPort` is the worker's side of the channel; `isMainThread` tells you which side you're on.

### Workers vs child processes vs cluster
| | Worker threads | Child processes (`fork`) | Cluster |
|-|----------------|-------------------------|---------|
| Isolation | Same process, separate heap | Separate process | Separate processes |
| Startup cost | ~ms, lower memory | Higher | Higher |
| Shared memory | Yes (`SharedArrayBuffer`) | No | No |
| Typical use | CPU tasks inside a server | Running other programs / isolating crashes | Scaling an HTTP server across cores |

### Don't create a worker per task
Starting a worker costs a few milliseconds and several MB of memory. For many small tasks, create a **pool** of `os.availableParallelism()` workers up front and queue tasks to them. Libraries like **Piscina** do this in production; you'll build a small one here.

### Errors
An uncaught exception inside a worker emits `'error'` on the `Worker` object and terminates that worker. The pool must handle this and replace it.

```js
// cpu.js
export function countPrimes(limit) {
  let count = 0;
  for (let n = 2; n < limit; n++) {
    let prime = true;
    for (let d = 2; d * d <= n; d++) if (n % d === 0) { prime = false; break; }
    if (prime) count++;
  }
  return count;
}
```

```js
// worker-server.js
import { createServer } from 'node:http';
import { Worker } from 'node:worker_threads';

function primesInWorker(limit) {
  return new Promise((resolve, reject) => {
    const w = new Worker(new URL('./prime-worker.js', import.meta.url), { workerData: { limit } });
    w.once('message', resolve);
    w.once('error', reject);
    w.once('exit', (code) => { if (code !== 0) reject(new Error(`worker exited ${code}`)); });
  });
}

createServer(async (req, res) => {
  if (req.url === '/primes') return res.end(`primes: ${await primesInWorker(15_000_000)}\n`);
  res.end('fast\n');
}).listen(3000);
```

### 3. Copy vs transfer

```js
// transfer.js
import { Worker } from 'node:worker_threads';

const worker = new Worker(`
  const { parentPort } = require('node:worker_threads');
  parentPort.on('message', (buf) => { buf[0] = 99; parentPort.postMessage(buf, [buf.buffer]); });
`, { eval: true });

const big = new Uint8Array(200 * 1024 * 1024);   // 200 MB

let t = performance.now();
worker.postMessage(big);                          // structured clone: copies 200 MB
console.log(`copy: ${(performance.now() - t).toFixed(1)}ms, still usable here: ${big.byteLength}`);
await new Promise((r) => worker.once('message', r));

t = performance.now();
worker.postMessage(big, [big.buffer]);            // transfer: moves ownership
console.log(`transfer: ${(performance.now() - t).toFixed(3)}ms, length here now: ${big.byteLength}`);   // 0
const back = await new Promise((r) => worker.once('message', r));
console.log('received back, first byte =', back[0]);
await worker.terminate();
```

### 4. Shared memory with Atomics

```js
// shared.js
import { Worker } from 'node:worker_threads';

const shared = new SharedArrayBuffer(4);
const counter = new Int32Array(shared);
const THREADS = 4, INCREMENTS = 1_000_000;

const code = `
  const { workerData } = require('node:worker_threads');
  const arr = new Int32Array(workerData.shared);
  for (let i = 0; i < workerData.n; i++) {
    if (workerData.atomic) Atomics.add(arr, 0, 1);
    else arr[0]++;                                   // read-modify-write race
  }
`;

for (const atomic of [false, true]) {
  counter[0] = 0;
  await Promise.all(Array.from({ length: THREADS }, () => new Promise((resolve) => {
    new Worker(code, { eval: true, workerData: { shared, n: INCREMENTS, atomic } }).on('exit', resolve);
  })));
  console.log(`atomic=${atomic}: expected ${THREADS * INCREMENTS}, got ${counter[0]}`);
}
```

### 5. Build a worker pool

```js
// pool.js
import { Worker } from 'node:worker_threads';
import { availableParallelism } from 'node:os';

export class WorkerPool {
  #file; #size; #idle = []; #queue = []; #all = new Set(); #closed = false;

  constructor(file, size = availableParallelism()) {
    this.#file = file;
    this.#size = size;
    for (let i = 0; i < size; i++) this.#spawn();
  }

  #spawn() {
    const worker = new Worker(this.#file);
    worker.on('message', ({ ok, value, error }) => {
      const { resolve, reject } = worker.current;
      worker.current = null;
      ok ? resolve(value) : reject(Object.assign(new Error(error.message), { stack: error.stack }));
      this.#release(worker);
    });
    worker.on('error', (err) => {                  // uncaught exception: worker is dead
      worker.current?.reject(err);
      worker.current = null;
      this.#all.delete(worker);
      this.#idle = this.#idle.filter((w) => w !== worker);   // it may have been idle when it crashed
      if (!this.#closed) this.#spawn();            // replace it
    });
    this.#all.add(worker);
    this.#release(worker);
  }

  #release(worker) {
    const next = this.#queue.shift();
    if (next) { worker.current = next; worker.postMessage(next.task); }
    else this.#idle.push(worker);
  }

  run(task) {
    if (this.#closed) return Promise.reject(new Error('pool is closed'));
    return new Promise((resolve, reject) => {
      const job = { task, resolve, reject };
      const worker = this.#idle.pop();
      if (worker) { worker.current = job; worker.postMessage(task); }
      else this.#queue.push(job);
    });
  }

  get pending() { return this.#queue.length; }

  async close() {
    this.#closed = true;
    await Promise.all([...this.#all].map((w) => w.terminate()));
  }
}
```

```js
// pool-worker.js
import { parentPort } from 'node:worker_threads';
import { countPrimes } from './cpu.js';

const handlers = {
  primes: ({ limit }) => countPrimes(limit),
  crash: () => { setImmediate(() => { throw new Error('uncaught in worker'); }); return 'scheduled crash'; },
  fail: () => { throw new Error('task failed'); },
};

parentPort.on('message', ({ type, payload }) => {
  try {
    parentPort.postMessage({ ok: true, value: handlers[type](payload) });
  } catch (err) {
    parentPort.postMessage({ ok: false, error: { message: err.message, stack: err.stack } });
  }
});
```

```js
// use-pool.js
import { availableParallelism } from 'node:os';
import { WorkerPool } from './pool.js';
import { countPrimes } from './cpu.js';

const tasks = Array.from({ length: 16 }, () => 3_000_000);

let t = performance.now();
tasks.forEach((limit) => countPrimes(limit));
console.log(`main thread: ${Math.round(performance.now() - t)}ms`);

const pool = new WorkerPool(new URL('./pool-worker.js', import.meta.url));
t = performance.now();
await Promise.all(tasks.map((limit) => pool.run({ type: 'primes', payload: { limit } })));
console.log(`pool of ${availableParallelism()}: ${Math.round(performance.now() - t)}ms`);

try { await pool.run({ type: 'fail' }); } catch (err) { console.log('task error:', err.message); }

await pool.run({ type: 'crash' });                         // worker dies right after replying
await new Promise((r) => setTimeout(r, 100));              // give the pool time to replace it
console.log('still works:', await Promise.all(
  Array.from({ length: 8 }, () => pool.run({ type: 'primes', payload: { limit: 1000 } })),
));
await pool.close();
```
