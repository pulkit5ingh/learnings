## Background

### Why plain hashes fail for passwords
SHA-256 is designed to be fast: a GPU can compute billions per second, so common passwords fall instantly. And identical passwords produce identical hashes, so one precomputed table (a "rainbow table") cracks every user at once.

### The fix: a slow, salted key derivation function
- **Salt** — a random value (16+ bytes) generated per password and stored alongside the hash. Identical passwords now produce different hashes, and precomputed tables are useless.
- **Work factor** — the function is deliberately expensive in CPU and memory, so each guess costs the attacker real time.

Node ships **scrypt** (memory-hard). `argon2id` (via the `argon2` package) and `bcrypt` are also good. Typical scrypt parameters: `N=2^15–2^17` (cost), `r=8`, `p=1`, 64-byte output — tune so one hash takes ~50–250 ms on your server. Note that `crypto.scrypt` runs on libuv's thread pool, so it doesn't block the event loop, but many concurrent logins can saturate the pool.

### Store parameters with the hash
Store a self-describing string such as `scrypt$N=32768,r=8,p=1$<salt>$<hash>`. When you later raise the cost, old hashes still verify with their own parameters, and you can **rehash on next successful login**.

### What a JWT is
A JSON Web Token is three base64url parts: `header.payload.signature`.
- **Header** — `{"alg":"HS256","typ":"JWT"}`
- **Payload** — claims: `sub` (subject), `iat` (issued at), `exp` (expiry), `iss`, `aud`, plus custom data. **It is encoded, not encrypted — anyone can read it.**
- **Signature** — `HMAC-SHA256(base64url(header) + "." + base64url(payload), secret)`.

### Classic JWT vulnerabilities
1. **`alg: none`** — accepting unsigned tokens. Always enforce the algorithm you expect; never trust the header's `alg`.
2. **Algorithm confusion** — accepting HS256 tokens signed with an RS256 **public** key as the HMAC secret.
3. **Missing `exp` checks** — tokens valid forever.
4. **Non-constant-time signature comparison.**
5. **Weak secrets** — HS256 secrets must be long random values (32+ bytes), not `"secret"`.

### 2. Hash and verify with scrypt

```js
// password.js
import { scrypt, randomBytes, timingSafeEqual } from 'node:crypto';
import { promisify } from 'node:util';

const scryptAsync = promisify(scrypt);
const DEFAULTS = { N: 2 ** 15, r: 8, p: 1, keylen: 64 };

export async function hashPassword(password, params = DEFAULTS) {
  const salt = randomBytes(16);
  const { N, r, p, keylen } = params;
  const key = await scryptAsync(password.normalize('NFKC'), salt, keylen, { N, r, p, maxmem: 256 * N * r });
  return `scrypt$N=${N},r=${r},p=${p}$${salt.toString('base64url')}$${key.toString('base64url')}`;
}

function parse(stored) {
  const [scheme, paramStr, saltB64, keyB64] = stored.split('$');
  if (scheme !== 'scrypt' || !keyB64) throw new Error('unrecognised hash format');
  const params = Object.fromEntries(paramStr.split(',').map(kv => {
    const [k, v] = kv.split('=');
    return [k, Number(v)];
  }));
  return { params, salt: Buffer.from(saltB64, 'base64url'), key: Buffer.from(keyB64, 'base64url') };
}

export async function verifyPassword(password, stored) {
  const { params: { N, r, p }, salt, key } = parse(stored);
  const candidate = await scryptAsync(password.normalize('NFKC'), salt, key.length, { N, r, p, maxmem: 256 * N * r });
  return timingSafeEqual(candidate, key);
}

export function needsRehash(stored, params = DEFAULTS) {
  const { params: current } = parse(stored);
  return current.N < params.N || current.r !== params.r || current.p !== params.p;
}
```

### 4. Build a JWT (HS256) by hand

```js
// jwt.js
import { createHmac, timingSafeEqual } from 'node:crypto';

const b64url = (obj) => Buffer.from(JSON.stringify(obj)).toString('base64url');
const hmac = (data, secret) => createHmac('sha256', secret).update(data).digest();

export function signJwt(payload, secret, { expiresInSec = 900, issuer } = {}) {
  const now = Math.floor(Date.now() / 1000);
  const header = { alg: 'HS256', typ: 'JWT' };
  const body = { iat: now, exp: now + expiresInSec, ...(issuer && { iss: issuer }), ...payload };
  const unsigned = `${b64url(header)}.${b64url(body)}`;
  return `${unsigned}.${hmac(unsigned, secret).toString('base64url')}`;
}

export class JwtError extends Error { name = 'JwtError'; }

export function verifyJwt(token, secret, { issuer, clockSkewSec = 30 } = {}) {
  const parts = token.split('.');
  if (parts.length !== 3) throw new JwtError('malformed token');
  const [h, p, s] = parts;

  let header, payload;
  try {
    header = JSON.parse(Buffer.from(h, 'base64url'));
    payload = JSON.parse(Buffer.from(p, 'base64url'));
  } catch {
    throw new JwtError('malformed token');
  }

  // Enforce the algorithm we expect — never trust header.alg to choose
  if (header.alg !== 'HS256') throw new JwtError(`unexpected alg ${header.alg}`);

  const expected = hmac(`${h}.${p}`, secret);
  const actual = Buffer.from(s, 'base64url');
  if (actual.length !== expected.length || !timingSafeEqual(actual, expected)) throw new JwtError('invalid signature');

  const now = Math.floor(Date.now() / 1000);
  if (typeof payload.exp !== 'number' || payload.exp + clockSkewSec < now) throw new JwtError('token expired');
  if (payload.nbf && payload.nbf - clockSkewSec > now) throw new JwtError('token not yet valid');
  if (issuer && payload.iss !== issuer) throw new JwtError('wrong issuer');
  return payload;
}
```

### 5. Attack your own implementation

```js
// attack.js
import { randomBytes } from 'node:crypto';
import { signJwt, verifyJwt } from './jwt.js';

const SECRET = randomBytes(32);
const token = signJwt({ sub: 'user_42', role: 'user' }, SECRET, { issuer: 'auth.example' });
console.log('token:', token);
console.log('payload is readable by anyone:', JSON.parse(Buffer.from(token.split('.')[1], 'base64url')));

const tryVerify = (label, t) => {
  try { console.log(label, '→ OK', verifyJwt(t, SECRET, { issuer: 'auth.example' }).sub); }
  catch (err) { console.log(label, '→ rejected:', err.message); }
};

tryVerify('valid', token);

// 1. Privilege escalation: edit the payload, keep the signature
const [h, , s] = token.split('.');
const forged = Buffer.from(JSON.stringify({ sub: 'user_42', role: 'admin', exp: 9999999999, iss: 'auth.example' })).toString('base64url');
tryVerify('edited payload', `${h}.${forged}.${s}`);

// 2. alg:none
const none = Buffer.from(JSON.stringify({ alg: 'none', typ: 'JWT' })).toString('base64url');
tryVerify('alg none', `${none}.${forged}.`);

// 3. Expired
tryVerify('expired', signJwt({ sub: 'x' }, SECRET, { expiresInSec: -120, issuer: 'auth.example' }));

// 4. Wrong secret / issuer
tryVerify('wrong secret', signJwt({ sub: 'x' }, 'secret', { issuer: 'auth.example' }));
tryVerify('wrong issuer', signJwt({ sub: 'x' }, SECRET, { issuer: 'evil.example' }));
```

Every attack must be rejected.
