# A7.2 — Mocking & Test Doubles

**Phase:** 7 — Testing & Debugging
**Topic:** `mock.fn`, `mock.method`, mock timers, `mock.module`, dependency injection, fakes vs stubs vs spies
**Type:** Assignment

---

## Goal
Test code that depends on time, the network, and other modules — quickly and deterministically — and design code so it is easy to test.

## Background

### Kinds of test doubles
| Double | What it is | Example |
|--------|-----------|---------|
| **Stub** | Returns canned answers | `fetchUser = async () => ({ id: 1 })` |
| **Spy** | Records calls for assertions | `t.mock.fn()` + `fn.mock.calls` |
| **Fake** | A working lightweight implementation | In-memory repository instead of a DB |
| **Module mock** | Replaces an imported module | `t.mock.module('./db.js', …)` |

Prefer **injected fakes** over patching modules — tests get clearer and code gets more modular.

### The node:test mocking API
Each test context has `t.mock`, restored automatically after the test.
- `t.mock.fn(impl?)` — spy; inspect `fn.mock.calls[i].arguments` and `fn.mock.callCount()`; change behaviour with `mockImplementation` / `mockImplementationOnce`.
- `t.mock.method(obj, 'name', impl?)` — replace a method on an existing object (e.g. `globalThis.fetch`).
- `t.mock.timers.enable({ apis: ['setTimeout', 'Date'] })` + `t.mock.timers.tick(ms)` — control time.
- `t.mock.module(specifier, { namedExports })` — needs `--experimental-test-module-mocks`, and the code under test must be imported **after** the mock (dynamic `import()`).

### Dependency injection without a framework
```js
export function createService({ fetch = globalThis.fetch, now = Date.now } = {}) { … }
```
Tests pass fakes; production passes nothing.

### What not to mock
Don't mock the code under test or simple pure helpers, and don't assert on every internal call. Test observable behaviour at the boundaries.

### 1. A service with dependencies

```bash

npm init -y && npm pkg set type=module
mkdir src test
```

```js
// src/weather.js
export class WeatherService {
  #fetch; #now; #ttlMs; #cache = new Map();

  constructor({ fetch = globalThis.fetch, now = Date.now, ttlMs = 60_000, apiUrl = 'https://api.example.com' } = {}) {
    this.#fetch = fetch; this.#now = now; this.#ttlMs = ttlMs; this.apiUrl = apiUrl;
  }

  async current(city) {
    const key = city.toLowerCase();
    const hit = this.#cache.get(key);
    if (hit && this.#now() - hit.at < this.#ttlMs) return hit.data;

    const res = await this.#fetch(`${this.apiUrl}/weather?city=${encodeURIComponent(city)}`);
    if (res.status === 404) throw new Error(`unknown city: ${city}`);
    if (!res.ok) throw new Error(`weather API error ${res.status}`);
    const data = await res.json();
    this.#cache.set(key, { at: this.#now(), data });
    return data;
  }
}
```

### 2. Inject spies and a fake clock

```js
// test/weather.test.js
import { describe, it } from 'node:test';
import assert from 'node:assert/strict';
import { WeatherService } from '../src/weather.js';

const json = (body, status = 200) => new Response(JSON.stringify(body), { status });

describe('WeatherService', () => {
  it('fetches weather data with the right URL', async (t) => {
    const fetch = t.mock.fn(async () => json({ tempC: 21 }));
    assert.deepEqual(await new WeatherService({ fetch }).current('São Paulo'), { tempC: 21 });
    assert.equal(fetch.mock.callCount(), 1);
    assert.equal(fetch.mock.calls[0].arguments[0], 'https://api.example.com/weather?city=S%C3%A3o%20Paulo');
  });

  it('caches until the TTL expires', async (t) => {
    let clock = 1_000_000;
    const fetch = t.mock.fn(async () => json({ tempC: 10 }));
    const svc = new WeatherService({ fetch, now: () => clock, ttlMs: 60_000 });
    await svc.current('Oslo');
    await svc.current('OSLO');
    assert.equal(fetch.mock.callCount(), 1);
    clock += 60_001;
    await svc.current('Oslo');
    assert.equal(fetch.mock.callCount(), 2);
  });

  it('maps 404 and other errors', async (t) => {
    const fetch = t.mock.fn();
    fetch.mock.mockImplementationOnce(async () => json({}, 404), 0);
    fetch.mock.mockImplementationOnce(async () => json({}, 503), 1);
    const svc = new WeatherService({ fetch });
    await assert.rejects(svc.current('Atlantis'), /unknown city/);
    await assert.rejects(svc.current('Paris'), /error 503/);
  });
});
```

### 3. Patch a global

```js
// test/global-fetch.test.js
import { it } from 'node:test';
import assert from 'node:assert/strict';
import { WeatherService } from '../src/weather.js';

it('uses globalThis.fetch by default', async (t) => {
  t.mock.method(globalThis, 'fetch', async () => new Response('{"tempC":5}'));
  // the default parameter reads globalThis.fetch at construction — after the mock is installed
  const svc = new WeatherService();
  assert.deepEqual(await svc.current('Reykjavik'), { tempC: 5 });
  assert.equal(globalThis.fetch.mock.callCount(), 1);
});
```

### 4. Mock timers and injected sleep

```js
// src/timing.js
export function debounce(fn, ms) {
  let timer;
  return (...args) => { clearTimeout(timer); timer = setTimeout(() => fn(...args), ms); };
}

export async function retry(fn, { attempts = 3, delayMs = 1000, sleep = (ms) => new Promise((r) => setTimeout(r, ms)) } = {}) {
  let lastErr;
  for (let i = 0; i < attempts; i++) {
    try { return await fn(i); }
    catch (err) { lastErr = err; if (i < attempts - 1) await sleep(delayMs * 2 ** i); }
  }
  throw lastErr;
}
```

```js
// test/timing.test.js
import { it } from 'node:test';
import assert from 'node:assert/strict';
import { debounce, retry } from '../src/timing.js';

it('debounce fires once after the quiet period', (t) => {
  t.mock.timers.enable({ apis: ['setTimeout'] });
  const fn = t.mock.fn();
  const d = debounce(fn, 300);
  d('a'); d('b');
  t.mock.timers.tick(299);
  assert.equal(fn.mock.callCount(), 0);
  t.mock.timers.tick(1);
  assert.deepEqual(fn.mock.calls.map((c) => c.arguments), [['b']]);
});

it('retry backs off exponentially without real waiting', async (t) => {
  const sleep = t.mock.fn(async () => {});
  const fn = t.mock.fn(async (i) => { if (i < 2) throw new Error(`fail ${i}`); return 'ok'; });
  assert.equal(await retry(fn, { delayMs: 100, sleep }), 'ok');
  assert.deepEqual(sleep.mock.calls.map((c) => c.arguments[0]), [100, 200]);
});
```

### 5. A fake repository

```js
// src/users.js
export function createUserService({ repo, hash, clock = () => new Date() }) {
  return {
    async register(email, password) {
      if (await repo.findByEmail(email)) throw new Error('email already registered');
      return repo.insert({ email: email.toLowerCase(), passwordHash: await hash(password), createdAt: clock().toISOString() });
    },
  };
}

export class InMemoryUserRepo {
  #rows = []; #id = 1;
  async findByEmail(email) { return this.#rows.find((r) => r.email === email.toLowerCase()) ?? null; }
  async insert(row) { const saved = { id: this.#id++, ...row }; this.#rows.push(saved); return saved; }
}
```

```js
// test/users.test.js
import { it } from 'node:test';
import assert from 'node:assert/strict';
import { createUserService, InMemoryUserRepo } from '../src/users.js';

it('registers users and rejects duplicates', async () => {
  const svc = createUserService({
    repo: new InMemoryUserRepo(),
    hash: async (p) => `hashed:${p}`,
    clock: () => new Date('2026-01-01T00:00:00Z'),
  });
  assert.deepEqual(await svc.register('Ada@Example.com', 'pw'),
    { id: 1, email: 'ada@example.com', passwordHash: 'hashed:pw', createdAt: '2026-01-01T00:00:00.000Z' });
  await assert.rejects(svc.register('ada@example.com', 'x'), /already registered/);
});
```

### 6. Module mocking when you can't inject

```js
// src/report.js
import { readFile } from 'node:fs/promises';
export async function lineCount(path) { return (await readFile(path, 'utf8')).split('\n').filter(Boolean).length; }
```

```js
// test/report.test.js
import { it } from 'node:test';
import assert from 'node:assert/strict';

it('counts non-empty lines', async (t) => {
  t.mock.module('node:fs/promises', { namedExports: { readFile: async () => 'a\nb\n\nc\n' } });
  const { lineCount } = await import('../src/report.js');   // import AFTER mocking
  assert.equal(await lineCount('/any/path'), 3);
});
```
