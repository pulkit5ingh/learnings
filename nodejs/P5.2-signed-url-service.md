# P5.2 — Signed URL Service

**Phase:** 5 — Security & Cryptography
**Topic:** HMAC-signed expiring URLs, canonicalisation, key rotation, one-time tokens, constant-time checks, safe file serving
**Type:** Project

---

## Goal

Build a file-sharing service that issues **signed, expiring download links** — the same idea as S3 pre-signed URLs or Cloudflare signed URLs. Anyone with the link can download the file until it expires; nobody can forge or extend a link without the server's secret.

## What to Build

- `POST /links` (requires `Authorization: Bearer <ADMIN_TOKEN>`) with `{ "file": "report.pdf", "ttlSeconds": 600, "oneTime": false, "downloadName": "Q3 report.pdf" }` → `{ url, expiresAt }`
- `GET /download/:file?exp=…&kid=…&once=…&nonce=…&sig=…` — serves the file only if the signature is valid and not expired
- Signature = HMAC-SHA256 over a **canonical string**: method, path, and sorted query parameters (excluding `sig`)
- Key rotation: multiple keys identified by `kid`; new links use the newest key, old keys still verify until removed
- One-time links: a random nonce is recorded on first use; reuse returns `410 Gone`
- Max TTL (24 h) and file name validation; files served only from `storage/`
- `Content-Disposition` with the requested download name (safely encoded)
- Constant-time signature comparison; precise errors: `403` bad signature, `410` expired or used, `404` missing file
- Access log without the signature value

## Architecture

```
signed-urls/
├── package.json
├── storage/                ← files available for sharing
└── src/
    ├── server.js
    ├── keys.js             ← load keys from env: SIGNING_KEYS="k2:hex,k1:hex"
    ├── signer.js           ← canonicalise, sign, verify
    ├── nonces.js           ← one-time token store with expiry
    └── serveFile.js        ← stream from storage/ safely
```

## Implementation

### Keys

```js
// src/keys.js
// SIGNING_KEYS="2026-10:<64 hex>,2026-07:<64 hex>" — first one is current
export function loadKeys(raw = process.env.SIGNING_KEYS) {
  if (!raw) throw new Error('SIGNING_KEYS is required, e.g. SIGNING_KEYS="k1:<64 hex chars>"');
  const keys = new Map();
  for (const entry of raw.split(',')) {
    const [kid, hex] = entry.trim().split(':');
    if (!kid || !/^[0-9a-f]{64}$/i.test(hex ?? '')) throw new Error(`invalid key entry "${kid}"`);
    keys.set(kid, Buffer.from(hex, 'hex'));
  }
  return { current: [...keys.keys()][0], keys };
}
```

Generate keys with `node -e "console.log(require('crypto').randomBytes(32).toString('hex'))"`.

### Signer

```js
// src/signer.js
import { createHmac, timingSafeEqual } from 'node:crypto';

export function canonical(method, pathname, params) {
  const sorted = [...params.entries()]
    .filter(([k]) => k !== 'sig')
    .sort(([a], [b]) => (a < b ? -1 : a > b ? 1 : 0))
    .map(([k, v]) => `${encodeURIComponent(k)}=${encodeURIComponent(v)}`)
    .join('&');
  return `${method.toUpperCase()}\n${pathname}\n${sorted}`;
}

const mac = (key, data) => createHmac('sha256', key).update(data).digest();

export function signUrl(baseUrl, pathname, params, { kid, key }) {
  const sp = new URLSearchParams({ ...params, kid });
  sp.set('sig', mac(key, canonical('GET', pathname, sp)).toString('base64url'));
  return `${baseUrl}${pathname}?${sp}`;
}

export class LinkError extends Error {
  constructor(status, message) { super(message); this.status = status; }
}

export function verifyUrl(method, url, keyring, now = Date.now()) {
  const sp = url.searchParams;
  const key = keyring.keys.get(sp.get('kid') ?? '');
  if (!key) throw new LinkError(403, 'unknown signing key');

  const given = Buffer.from(sp.get('sig') ?? '', 'base64url');
  const expected = mac(key, canonical(method, url.pathname, sp));
  if (given.length !== expected.length || !timingSafeEqual(given, expected)) throw new LinkError(403, 'invalid signature');

  // Only trust parameters AFTER the signature check passes
  const exp = Number(sp.get('exp'));
  if (!Number.isInteger(exp) || exp * 1000 < now) throw new LinkError(410, 'link expired');
  return Object.fromEntries(sp);
}
```

Signing the method and path prevents a signature for one file being reused for another; sorting parameters makes the signature independent of their order.

### One-time nonces

```js
// src/nonces.js
export class NonceStore {
  #used = new Map();     // nonce → expiry ms

  constructor() {
    setInterval(() => {
      const now = Date.now();
      for (const [n, exp] of this.#used) if (exp < now) this.#used.delete(n);
    }, 60_000).unref();   // cleanup must not keep the process alive
  }

  // Returns true the first time a nonce is seen, false afterwards. Atomic within one process.
  consume(nonce, expiresAtMs) {
    if (this.#used.has(nonce)) return false;
    this.#used.set(nonce, expiresAtMs);
    return true;
  }
}
```

In a multi-instance deployment this would be Redis `SET nonce 1 NX PX ttl` — note the limitation in your README.

### Serving files

```js
// src/serveFile.js
import { createReadStream } from 'node:fs';
import { stat } from 'node:fs/promises';
import { resolve, sep, extname } from 'node:path';
import { pipeline } from 'node:stream/promises';

export const STORAGE = resolve(import.meta.dirname, '..', 'storage');
export const SAFE_NAME = /^[\w][\w.-]{0,127}$/;     // no slashes, no leading dot

const TYPES = { '.pdf': 'application/pdf', '.txt': 'text/plain; charset=utf-8', '.png': 'image/png', '.zip': 'application/zip' };

function contentDisposition(name) {
  const ascii = name.replace(/[^\x20-\x7e]/g, '_').replace(/["\\]/g, '_');
  return `attachment; filename="${ascii}"; filename*=UTF-8''${encodeURIComponent(name)}`;
}

export async function fileExists(name) {
  if (!SAFE_NAME.test(name)) return false;
  try { return (await stat(resolve(STORAGE, name))).isFile(); } catch { return false; }
}

export async function sendFile(res, name, downloadName) {
  const full = resolve(STORAGE, name);
  if (!SAFE_NAME.test(name) || !full.startsWith(STORAGE + sep)) throw Object.assign(new Error('bad name'), { status: 400 });
  const s = await stat(full);
  res.writeHead(200, {
    'content-type': TYPES[extname(name)] ?? 'application/octet-stream',
    'content-length': s.size,
    'content-disposition': contentDisposition(downloadName || name),
    'cache-control': 'private, no-store',
    'x-content-type-options': 'nosniff',
  });
  await pipeline(createReadStream(full), res);
}
```

### Server

```js
// src/server.js
import { createServer } from 'node:http';
import { randomBytes, timingSafeEqual, createHash } from 'node:crypto';
import { loadKeys } from './keys.js';
import { signUrl, verifyUrl, LinkError } from './signer.js';
import { NonceStore } from './nonces.js';
import { sendFile, fileExists, SAFE_NAME } from './serveFile.js';

const keyring = loadKeys();
const nonces = new NonceStore();
const ADMIN_TOKEN = process.env.ADMIN_TOKEN ?? '';
const BASE_URL = process.env.BASE_URL ?? 'http://localhost:3000';
const MAX_TTL = 24 * 3600;

const sha = (s) => createHash('sha256').update(s).digest();
const isAdmin = (req) => ADMIN_TOKEN.length >= 16 && timingSafeEqual(sha(req.headers.authorization ?? ''), sha(`Bearer ${ADMIN_TOKEN}`));

const json = (res, status, body) => { res.writeHead(status, { 'content-type': 'application/json' }); res.end(JSON.stringify(body)); };

async function readJson(req, limit = 10_000) {
  let size = 0; const chunks = [];
  for await (const c of req) { if ((size += c.length) > limit) throw new LinkError(413, 'body too large'); chunks.push(c); }
  try { return JSON.parse(Buffer.concat(chunks)); } catch { throw new LinkError(400, 'invalid JSON'); }
}

async function createLink(req, res) {
  if (!isAdmin(req)) throw new LinkError(401, 'unauthorized');
  const { file, ttlSeconds = 600, oneTime = false, downloadName } = await readJson(req);
  if (typeof file !== 'string' || !SAFE_NAME.test(file)) throw new LinkError(400, 'invalid file name');
  if (!(await fileExists(file))) throw new LinkError(404, 'no such file');
  if (!Number.isInteger(ttlSeconds) || ttlSeconds < 1 || ttlSeconds > MAX_TTL) throw new LinkError(400, `ttlSeconds must be 1–${MAX_TTL}`);

  const exp = Math.floor(Date.now() / 1000) + ttlSeconds;
  const params = { exp: String(exp) };
  if (oneTime) { params.once = '1'; params.nonce = randomBytes(16).toString('base64url'); }
  if (downloadName) params.dn = String(downloadName).slice(0, 200);

  const url = signUrl(BASE_URL, `/download/${file}`, params, { kid: keyring.current, key: keyring.keys.get(keyring.current) });
  json(res, 201, { url, expiresAt: new Date(exp * 1000).toISOString() });
}

async function download(req, res, url) {
  const claims = verifyUrl(req.method === 'HEAD' ? 'GET' : req.method, url, keyring);
  const file = decodeURIComponent(url.pathname.slice('/download/'.length));
  if (claims.once === '1' && !nonces.consume(claims.nonce, Number(claims.exp) * 1000)) throw new LinkError(410, 'link already used');
  if (!(await fileExists(file))) throw new LinkError(404, 'file no longer exists');
  await sendFile(res, file, claims.dn);
}

createServer(async (req, res) => {
  const url = new URL(req.url, BASE_URL);
  const started = performance.now();
  res.on('finish', () => console.log(`${req.method} ${url.pathname} ${res.statusCode} ${(performance.now() - started).toFixed(1)}ms`)); // no sig in logs
  try {
    if (req.method === 'POST' && url.pathname === '/links') return await createLink(req, res);
    if (req.method === 'GET' && url.pathname.startsWith('/download/')) return await download(req, res, url);
    json(res, 404, { error: 'not found' });
  } catch (err) {
    if (err.code === 'ERR_STREAM_PREMATURE_CLOSE') return;
    const status = err.status ?? 500;
    if (status === 500) console.error(err);
    if (!res.headersSent) json(res, status, { error: status === 500 ? 'internal error' : err.message });
  }
}).listen(3000, () => console.log(`signed URL service on ${BASE_URL} (current key: ${keyring.current})`));
```

### Try it

```bash
mkdir -p storage && echo "quarterly numbers" > storage/report.txt
export SIGNING_KEYS="k2:$(openssl rand -hex 32),k1:$(openssl rand -hex 32)" ADMIN_TOKEN=$(openssl rand -hex 16)
node src/server.js

URL=$(curl -s -X POST localhost:3000/links -H "authorization: Bearer $ADMIN_TOKEN" \
  -d '{"file":"report.txt","ttlSeconds":30,"downloadName":"Q3 résumé.txt"}' | jq -r .url)
curl -i "$URL"                                              # 200 + content-disposition
curl -i "${URL/exp=/exp=9}"                                 # tampered expiry → 403
curl -i "${URL/report.txt/secret.txt}"                      # different file → 403
sleep 31 && curl -i "$URL"                                  # 410 expired

ONCE=$(curl -s -X POST localhost:3000/links -H "authorization: Bearer $ADMIN_TOKEN" -d '{"file":"report.txt","oneTime":true}' | jq -r .url)
curl -s -o /dev/null -w '%{http_code}\n' "$ONCE"            # 200
curl -s -o /dev/null -w '%{http_code}\n' "$ONCE"            # 410
curl -i -X POST localhost:3000/links -d '{"file":"report.txt"}'   # 401
```

Rotate keys: restart with `SIGNING_KEYS="k3:…,k2:…"` — links signed with `k2` still work, `k1` links are rejected.

## Deliverables

- [ ] Links verify only with the exact method, path, and parameters that were signed
- [ ] Changing `exp`, the file name, or any parameter yields `403`
- [ ] Expired links return `410`; one-time links return `410` on reuse
- [ ] Parameters are trusted only after the signature check passes
- [ ] Signatures and the admin token are compared in constant time
- [ ] Key rotation works via `kid`; removed keys invalidate their links
- [ ] File names are validated and confined to `storage/`
- [ ] `Content-Disposition` is safe for non-ASCII and quote characters
- [ ] Logs never contain the `sig` value
